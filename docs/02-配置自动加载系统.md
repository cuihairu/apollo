## 1. 系统概述

### 1.1 设计目标
- **热更新**：无需重启服务器即可更新配置
- **版本控制**：支持配置回滚和版本管理
- **校验机制**：自动校验配置的完整性和正确性
- **缓存优化**：提高配置访问性能
- **依赖管理**：处理配置间的依赖关系

#### ConfigManager实现
```cpp
// ConfigManager.h
class ConfigManager {
    struct ConfigInfo {
        std::string fileName;        // 文件名
        std::string filePath;        // 文件路径
        ConfigType type;             // 配置类型
        std::string version;         // 版本号
        uint64_t lastModified;       // 最后修改时间
        std::vector<std::string> dependencies;  // 依赖的配置
        bool isLoaded;              // 是否已加载
        std::weak_ptr<IConfigLoader> loader;    // 加载器
    };

    using ConfigMap = std::unordered_map<std::string, ConfigInfo>;
    ConfigMap configs;
    std::mutex configMutex;

    // 文件监控器
    FileWatcher fileWatcher;
    // 配置缓存
    ConfigCache configCache;
    // 版本管理器
    ConfigVersionManager versionManager;

public:
    static ConfigManager* Instance() {
        static ConfigManager instance;
        return &instance;
    }

    // 初始化配置系统
    bool Initialize(const std::string& configRoot) {
        // 1. 扫描配置文件
        if (!ScanConfigFiles(configRoot)) {
            return false;
        }

        // 2. 加载配置依赖关系
        if (!LoadDependencies()) {
            return false;
        }

        // 3. 启动文件监控
        fileWatcher.StartWatching(configRoot,
            std::bind(&ConfigManager::OnFileChanged, this,
                     std::placeholders::_1));

        // 4. 加载所有配置
        return LoadAllConfigs();
    }

    // 注册配置加载器
    template<typename T>
    void RegisterLoader(const std::string& pattern) {
        auto loader = std::make_shared<ConfigLoader<T>>();
        auto patterns = SplitString(pattern, ";");

        for (const auto& pat : patterns) {
            loaders[pat] = loader;
        }
    }

    // 获取配置
    template<typename T>
    std::shared_ptr<T> GetConfig(const std::string& name) {
        std::lock_guard<std::mutex> lock(configMutex);

        auto it = configs.find(name);
        if (it == configs.end() || !it->second.isLoaded) {
            LOG_ERROR("Config not found or not loaded: %s", name.c_str());
            return nullptr;
        }

        // 尝试从缓存获取
        auto cached = configCache.Get<T>(name);
        if (cached) {
            return cached;
        }

        // 加载配置
        return LoadConfig<T>(name);
    }

private:
    std::unordered_map<std::string, std::shared_ptr<IConfigLoader>> loaders;

    // 扫描配置文件
    bool ScanConfigFiles(const std::string& root) {
        std::vector<std::string> files;
        if (!FileSystem::ListFilesRecursive(root, files)) {
            return false;
        }

        for (const auto& file : files) {
            ConfigInfo info;
            info.fileName = FileSystem::GetFileName(file);
            info.filePath = file;
            info.type = GetConfigType(file);
            info.lastModified = FileSystem::GetLastModified(file);
            info.isLoaded = false;

            configs[info.fileName] = info;
        }

        return true;
    }

    // 加载所有配置
    bool LoadAllConfigs() {
        // 按依赖关系排序
        auto sortedConfigs = TopologicalSort();

        for (const auto& configName : sortedConfigs) {
            if (!LoadConfigByName(configName)) {
                LOG_ERROR("Failed to load config: %s", configName.c_str());
                return false;
            }
        }

        return true;
    }

    // 文件变更处理
    void OnFileChanged(const std::string& filePath) {
        auto fileName = FileSystem::GetFileName(filePath);
        auto it = configs.find(fileName);

        if (it == configs.end()) {
            return;
        }

        ConfigInfo& info = it->second;

        // 检查文件修改时间
        uint64_t newModified = FileSystem::GetLastModified(filePath);
        if (newModified <= info.lastModified) {
            return;  // 没有实际修改
        }

        LOG_INFO("Config file changed: %s", fileName.c_str());
        info.lastModified = newModified;

        // 重新加载配置
        ReloadConfig(fileName);
    }

    // 重新加载配置
    bool ReloadConfig(const std::string& name) {
        auto it = configs.find(name);
        if (it == configs.end()) {
            return false;
        }

        // 检查依赖的配置是否也需要重新加载
        std::vector<std::string> reloadList;
        ReloadDependencies(name, reloadList);

        // 按顺序重新加载
        for (const auto& configName : reloadList) {
            if (!LoadConfigByName(configName)) {
                LOG_ERROR("Failed to reload config: %s", configName.c_str());
                return false;
            }

            // 通知配置变更
            NotifyConfigChanged(configName);
        }

        return true;
    }
};
```

### 2.3 配置加载器接口

#### IConfigLoader定义
```cpp
// IConfigLoader.h
class IConfigLoader {
public:
    virtual ~IConfigLoader() = default;

    // 加载配置
    virtual bool Load(const std::string& filePath) = 0;

    // 重新加载
    virtual bool Reload(const std::string& filePath) = 0;

    // 校验配置
    virtual bool Validate(const std::string& filePath) = 0;

    // 获取配置数据
    virtual void* GetData() = 0;

    // 获取配置类型
    virtual const std::type_info& GetType() = 0;
};

// 模板化配置加载器
template<typename T>
class ConfigLoader : public IConfigLoader {
    std::shared_ptr<T> data;
    std::string lastHash;

public:
    bool Load(const std::string& filePath) override {
        try {
            // 读取文件
            std::string content;
            if (!FileSystem::ReadFile(filePath, content)) {
                return false;
            }

            // 解析配置
            data = ParseConfig(content);
            if (!data) {
                return false;
            }

            // 计算哈希
            lastHash = CalculateHash(content);

            return true;
        } catch (const std::exception& e) {
            LOG_ERROR("Failed to load config: %s, error: %s",
                     filePath.c_str(), e.what());
            return false;
        }
    }

    bool Reload(const std::string& filePath) override {
        // 检查内容是否变化
        std::string content;
        if (!FileSystem::ReadFile(filePath, content)) {
            return false;
        }

        std::string newHash = CalculateHash(content);
        if (newHash == lastHash) {
            return true;  // 内容没有变化
        }

        // 创建新配置
        auto newData = ParseConfig(content);
        if (!newData) {
            return false;
        }

        // 原子性替换
        data = newData;
        lastHash = newHash;

        return true;
    }

    bool Validate(const std::string& filePath) override {
        // 基础校验
        if (!FileSystem::FileExists(filePath)) {
            return false;
        }

        // 加载并校验
        std::string content;
        if (!FileSystem::ReadFile(filePath, content)) {
            return false;
        }

        auto tempData = ParseConfig(content);
        if (!tempData) {
            return false;
        }

        // 执行详细校验
        return ValidateConfig(*tempData);
    }

    void* GetData() override {
        return data.get();
    }

    const std::type_info& GetType() override {
        return typeid(T);
    }

protected:
    // 解析配置（子类实现）
    virtual std::shared_ptr<T> ParseConfig(const std::string& content) = 0;

    // 校验配置（子类实现）
    virtual bool ValidateConfig(const T& config) = 0;

private:
    std::string CalculateHash(const std::string& content) {
        return MD5(content).hexdigest();
    }
};
```

## 3. Excel到Lua转换

### 3.1 转换工具设计

#### Xls2Lua工具
```cpp
// Xls2Lua.cpp
class Xls2LuaConverter {
    struct ColumnInfo {
        std::string name;         // 列名
        std::string type;         // 数据类型
        bool isPrimaryKey;        // 是否主键
        bool isArray;             // 是否数组
        std::string defaultValue; // 默认值
        std::string comment;      // 注释
    };

    struct SheetInfo {
        std::string name;         // 表名
        std::vector<ColumnInfo> columns;  // 列信息
        int startRow;             // 数据起始行
        std::string outputFormat; // 输出格式
    };

public:
    // 转换Excel文件
    bool ConvertExcel(const std::string& excelPath,
                     const std::string& outputDir) {
        // 1. 打开Excel文件
        ExcelReader reader;
        if (!reader.Open(excelPath)) {
            LOG_ERROR("Failed to open Excel file: %s", excelPath.c_str());
            return false;
        }

        // 2. 解析元数据
        if (!ParseMetadata(reader)) {
            return false;
        }

        // 3. 读取数据
        if (!ReadData(reader)) {
            return false;
        }

        // 4. 生成Lua文件
        return GenerateLuaFiles(outputDir);
    }

    // 批量转换
    bool ConvertDirectory(const std::string& excelDir,
                         const std::string& outputDir) {
        std::vector<std::string> excelFiles;
        if (!FileSystem::ListFiles(excelDir, "*.xlsx", excelFiles)) {
            return false;
        }

        for (const auto& file : excelFiles) {
            if (!ConvertExcel(file, outputDir)) {
                LOG_ERROR("Failed to convert: %s", file.c_str());
                continue;
            }
        }

        return true;
    }

private:
    std::vector<SheetInfo> sheets;
    std::map<std::string, std::vector<std::map<std::string, std::string>>> dataMap;

    // 解析元数据
    bool ParseMetadata(ExcelReader& reader) {
        for (const auto& sheetName : reader.GetSheetNames()) {
            SheetInfo sheet;
            sheet.name = sheetName;
            sheet.startRow = 3;  // 默认从第3行开始是数据

            // 读取表头信息（第1行）
            if (!ReadColumnHeaders(reader, sheetName, sheet)) {
                return false;
            }

            // 读取类型信息（第2行）
            if (!ReadColumnTypes(reader, sheetName, sheet)) {
                return false;
            }

            sheets.push_back(sheet);
        }

        return true;
    }

    // 读取列头
    bool ReadColumnHeaders(ExcelReader& reader,
                           const std::string& sheetName,
                           SheetInfo& sheet) {
        auto headers = reader.GetRow(sheetName, 1);
        if (headers.empty()) {
            return false;
        }

        for (size_t i = 0; i < headers.size(); ++i) {
            ColumnInfo col;
            col.name = headers[i];
            col.isPrimaryKey = (headers[i].find("*") != std::string::npos);
            col.isArray = (headers[i].find("[]") != std::string::npos);

            // 清理特殊标记
            col.name = std::regex_replace(col.name, std::regex("\\*|\\[]"), "");

            if (!col.name.empty()) {
                sheet.columns.push_back(col);
            }
        }

        return !sheet.columns.empty();
    }

    // 读取数据
    bool ReadData(ExcelReader& reader) {
        for (const auto& sheet : sheets) {
            std::vector<std::map<std::string, std::string>> sheetData;

            for (int row = sheet.startRow; ; ++row) {
                auto rowData = reader.GetRow(sheet.name, row);
                if (rowData.empty()) {
                    break;  // 空行，数据结束
                }

                std::map<std::string, std::string> record;
                for (size_t i = 0; i < sheet.columns.size() && i < rowData.size(); ++i) {
                    record[sheet.columns[i].name] = rowData[i];
                }

                sheetData.push_back(record);
            }

            dataMap[sheet.name] = sheetData;
        }

        return true;
    }

    // 生成Lua文件
    bool GenerateLuaFiles(const std::string& outputDir) {
        for (const auto& sheet : sheets) {
            std::string outputPath = outputDir + "/" + sheet.name + ".lua";
            std::ofstream luaFile(outputPath);

            if (!luaFile.is_open()) {
                LOG_ERROR("Failed to create Lua file: %s", outputPath.c_str());
                return false;
            }

            // 生成文件头
            GenerateFileHeader(luaFile, sheet);

            // 生成数据表
            GenerateDataTable(luaFile, sheet);

            // 生成索引函数
            GenerateIndexFunctions(luaFile, sheet);

            // 生成查询函数
            GenerateQueryFunctions(luaFile, sheet);

            luaFile.close();
            LOG_INFO("Generated Lua file: %s", outputPath.c_str());
        }

        return true;
    }

    // 生成数据表
    void GenerateDataTable(std::ofstream& file, const SheetInfo& sheet) {
        file << "-- " << sheet.name << " 数据表\n";
        file << "local " << sheet.name << " = {\n";

        const auto& data = dataMap[sheet.name];
        for (const auto& record : data) {
            file << "    {";
            bool first = true;

            for (const auto& col : sheet.columns) {
                if (!first) file << ", ";
                first = false;

                auto it = record.find(col.name);
                if (it != record.end()) {
                    file << col.name << " = " << FormatLuaValue(it->second, col.type);
                }
            }

            file << "},\n";
        }

        file << "}\n\n";
    }

    // 格式化Lua值
    std::string FormatLuaValue(const std::string& value, const std::string& type) {
        if (value.empty()) return "nil";

        if (type == "string" || type == "text") {
            return "\"" + EscapeLuaString(value) + "\"";
        } else if (type == "number" || type == "int" || type == "float") {
            return value;
        } else if (type == "bool") {
            return (value == "1" || value == "true") ? "true" : "false";
        } else if (type == "table" || type.find("[]") != std::string::npos) {
            // 处理数组或表
            return FormatLuaArray(value);
        } else {
            return "\"" + EscapeLuaString(value) + "\"";
        }
    }

    // 转义Lua字符串
    std::string EscapeLuaString(const std::string& str) {
        std::string result;
        for (char c : str) {
            switch (c) {
                case '\"': result += "\\\""; break;
                case '\'': result += "\\\'"; break;
                case '\\': result += "\\\\"; break;
                case '\n': result += "\\n"; break;
                case '\r': result += "\\r"; break;
                case '\t': result += "\\t"; break;
                default: result += c; break;
            }
        }
        return result;
    }
};
```

### 3.2 Lua配置加载器

#### LuaConfigLoader实现
```cpp
// LuaConfigLoader.h
class LuaConfigLoader : public ConfigLoader<LuaConfig> {
    lua_State* L;

public:
    LuaConfigLoader() : L(nullptr) {
        L = luaL_newstate();
        luaL_openlibs(L);
    }

    ~LuaConfigLoader() {
        if (L) {
            lua_close(L);
        }
    }

protected:
    std::shared_ptr<LuaConfig> ParseConfig(const std::string& content) override {
        auto config = std::make_shared<LuaConfig>();

        // 执行Lua脚本
        if (luaL_dostring(L, content.c_str()) != 0) {
            LOG_ERROR("Lua error: %s", lua_tostring(L, -1));
            return nullptr;
        }

        // 提取配置表
        lua_getglobal(L, "config");
        if (!lua_istable(L, -1)) {
            LOG_ERROR("Config table not found");
            return nullptr;
        }

        // 解析配置表
        ParseLuaTable(L, -1, config->data);

        lua_pop(L, 1);
        return config;
    }

    bool ValidateConfig(const LuaConfig& config) override {
        // 检查必要的配置项
        if (!config.Has("server_id")) {
            LOG_ERROR("Missing server_id in config");
            return false;
        }

        if (!config.Has("port")) {
            LOG_ERROR("Missing port in config");
            return false;
        }

        // 类型校验
        if (!config.IsNumber("port")) {
            LOG_ERROR("Port must be a number");
            return false;
        }

        // 范围校验
        int port = config.GetInt("port");
        if (port < 1024 || port > 65535) {
            LOG_ERROR("Invalid port range: %d", port);
            return false;
        }

        return true;
    }

private:
    void ParseLuaTable(lua_State* L, int index,
                      std::map<std::string, ConfigValue>& result) {
        lua_pushnil(L);  // 第一个key

        while (lua_next(L, index) != 0) {
            std::string key;
            ConfigValue value;

            // 获取key
            if (lua_isstring(L, -2)) {
                key = lua_tostring(L, -2);
            } else {
                lua_pop(L, 1);
                continue;
            }

            // 获取value
            if (lua_isnumber(L, -1)) {
                value.type = ConfigValue::NUMBER;
                value.numberValue = lua_tonumber(L, -1);
            } else if (lua_isstring(L, -1)) {
                value.type = ConfigValue::STRING;
                value.stringValue = lua_tostring(L, -1);
            } else if (lua_isboolean(L, -1)) {
                value.type = ConfigValue::BOOLEAN;
                value.boolValue = lua_toboolean(L, -1);
            } else if (lua_istable(L, -1)) {
                value.type = ConfigValue::TABLE;
                ParseLuaTable(L, -1, value.tableValue);
            } else {
                value.type = ConfigValue::NIL;
            }

            result[key] = value;
            lua_pop(L, 1);  // 移除value，保留key用于下次迭代
        }
    }
};
```

## 4. 配置热更新机制

### 4.1 文件监控器

#### FileWatcher实现
```cpp
// FileWatcher.h
class FileWatcher {
#ifdef _WIN32
    HANDLE dirHandle;
    std::vector<BYTE> buffer;
    OVERLAPPED overlapped;
#else
    int inotifyFd;
    std::unordered_map<int, std::string> watchDescriptors;
#endif

    std::thread watchThread;
    std::atomic<bool> running{false};
    std::function<void(const std::string&)> changeCallback;

public:
    FileWatcher() : buffer(4096) {
#ifdef _WIN32
        dirHandle = INVALID_HANDLE_VALUE;
        memset(&overlapped, 0, sizeof(overlapped));
        overlapped.hEvent = CreateEvent(NULL, TRUE, FALSE, NULL);
#endif
    }

    ~FileWatcher() {
        Stop();
    }

    // 开始监控目录
    bool StartWatching(const std::string& directory,
                      std::function<void(const std::string&)> callback) {
        changeCallback = callback;

#ifdef _WIN32
        dirHandle = CreateFileA(
            directory.c_str(),
            FILE_LIST_DIRECTORY,
            FILE_SHARE_READ | FILE_SHARE_WRITE | FILE_SHARE_DELETE,
            NULL,
            OPEN_EXISTING,
            FILE_FLAG_BACKUP_SEMANTICS | FILE_FLAG_OVERLAPPED,
            NULL
        );

        if (dirHandle == INVALID_HANDLE_VALUE) {
            LOG_ERROR("Failed to open directory handle");
            return false;
        }
#else
        inotifyFd = inotify_init();
        if (inotifyFd == -1) {
            LOG_ERROR("Failed to initialize inotify");
            return false;
        }

        int wd = inotify_add_watch(inotifyFd, directory.c_str(),
                                  IN_MODIFY | IN_CREATE | IN_DELETE);
        if (wd == -1) {
            LOG_ERROR("Failed to add watch");
            return false;
        }

        watchDescriptors[wd] = directory;
#endif

        running = true;
        watchThread = std::thread(&FileWatcher::WatchLoop, this);
        return true;
    }

    void Stop() {
        running = false;

        if (watchThread.joinable()) {
            watchThread.join();
        }

#ifdef _WIN32
        if (dirHandle != INVALID_HANDLE_VALUE) {
            CloseHandle(dirHandle);
            dirHandle = INVALID_HANDLE_VALUE;
        }
        CloseHandle(overlapped.hEvent);
#else
        if (inotifyFd != -1) {
            for (const auto& pair : watchDescriptors) {
                inotify_rm_watch(inotifyFd, pair.first);
            }
            close(inotifyFd);
            inotifyFd = -1;
        }
#endif
    }

private:
    void WatchLoop() {
        while (running) {
#ifdef _WIN32
            DWORD bytesReturned;
            if (ReadDirectoryChangesW(
                dirHandle,
                buffer.data(),
                buffer.size(),
                TRUE,
                FILE_NOTIFY_CHANGE_FILE_NAME | FILE_NOTIFY_CHANGE_SIZE |
                FILE_NOTIFY_CHANGE_LAST_WRITE | FILE_NOTIFY_CHANGE_CREATION,
                &bytesReturned,
                &overlapped,
                NULL)) {

                WaitForSingleObject(overlapped.hEvent, INFINITE);
                ProcessWindowsChanges(bytesReturned);
                ResetEvent(overlapped.hEvent);
            }
#else
            fd_set fds;
            FD_ZERO(&fds);
            FD_SET(inotifyFd, &fds);

            struct timeval timeout;
            timeout.tv_sec = 1;
            timeout.tv_usec = 0;

            if (select(inotifyFd + 1, &fds, NULL, NULL, &timeout) > 0) {
                ProcessLinuxChanges();
            }
#endif
        }
    }

#ifdef _WIN32
    void ProcessWindowsChanges(DWORD bytesReturned) {
        if (bytesReturned == 0) return;

        FILE_NOTIFY_INFORMATION* info =
            reinterpret_cast<FILE_NOTIFY_INFORMATION*>(buffer.data());

        while (true) {
            std::wstring fileName(info->FileName,
                                 info->FileNameLength / sizeof(WCHAR));
            std::string utf8FileName = WStringToUTF8(fileName);

            switch (info->Action) {
                case FILE_ACTION_MODIFIED:
                case FILE_ACTION_RENAMED_NEW_NAME:
                    if (changeCallback) {
                        changeCallback(utf8FileName);
                    }
                    break;
            }

            if (info->NextEntryOffset == 0) break;
            info = reinterpret_cast<FILE_NOTIFY_INFORMATION*>(
                reinterpret_cast<BYTE*>(info) + info->NextEntryOffset);
        }
    }
#else
    void ProcessLinuxChanges() {
        const int BUFFER_SIZE = 4096;
        char buffer[BUFFER_SIZE];

        ssize_t length = read(inotifyFd, buffer, BUFFER_SIZE);
        if (length < 0) return;

        int i = 0;
        while (i < length) {
            struct inotify_event* event =

            }
```

```cpp
// HotReloadController.h
class HotReloadController {
    struct ReloadTask {
        std::string configName;
        std::string filePath;
        ReloadReason reason;
        uint64_t timestamp;
    };

    ThreadSafeQueue<ReloadTask> reloadQueue;
    std::thread reloadThread;
    std::atomic<bool> running{false};

    // 重载监听器
    std::vector<std::function<void(const std::string&, bool)>> listeners;

    // 重载统计
    std::atomic<uint64_t> totalReloads{0};
    std::atomic<uint64_t> successfulReloads{0};
    std::atomic<uint64_t> failedReloads{0};

public:
    enum ReloadReason {
        FILE_CHANGED,
        DEPENDENCY_CHANGED,
        MANUAL_TRIGGER,
        SCHEDULED_RELOAD
    };

    HotReloadController() {
        running = true;
        reloadThread = std::thread(&HotReloadController::ReloadLoop, this);
    }

    ~HotReloadController() {
        running = false;
        if (reloadThread.joinable()) {
            reloadThread.join();
        }
    }

    // 添加重载任务
    void ScheduleReload(const std::string& configName,
                       const std::string& filePath,
                       ReloadReason reason) {
        ReloadTask task;
        task.configName = configName;
        task.filePath = filePath;
        task.reason = reason;
        task.timestamp = GetCurrentTime();

        reloadQueue.push(task);
    }

    // 手动触发重载
    bool TriggerReload(const std::string& configName) {
        auto config = ConfigManager::Instance()->GetConfigInfo(configName);
        if (!config) {
            LOG_ERROR("Config not found: %s", configName.c_str());
            return false;
        }

        ScheduleReload(configName, config->filePath, MANUAL_TRIGGER);
        return true;
    }

    // 添加监听器
    void AddListener(std::function<void(const std::string&, bool)> listener) {
        listeners.push_back(listener);
    }

    // 获取统计信息
    ReloadStats GetStats() const {
        ReloadStats stats;
        stats.total = totalReloads.load();
        stats.successful = successfulReloads.load();
        stats.failed = failedReloads.load();
        stats.successRate = stats.total > 0 ?
            (double)stats.successful / stats.total : 0.0;
        return stats;
    }

private:
    void ReloadLoop() {
        while (running) {
            std::vector<ReloadTask> tasks;
            reloadQueue.try_pop_bulk(tasks, 10);  // 批量处理10个任务

            if (tasks.empty()) {
                std::this_thread::sleep_for(std::chrono::milliseconds(100));
                continue;
            }

            // 按优先级排序
            std::sort(tasks.begin(), tasks.end(),
                [](const ReloadTask& a, const ReloadTask& b) {
                    return a.reason < b.reason;  // 手动触发优先
                });

            // 处理重载任务
            for (const auto& task : tasks) {
                ProcessReload(task);
            }
        }
    }

    void ProcessReload(const ReloadTask& task) {
        totalReloads++;

        LOG_INFO("Reloading config: %s (reason: %d)",
                 task.configName.c_str(), task.reason);

        // 执行重载
        bool success = ConfigManager::Instance()->ReloadConfig(task.configName);

        // 更新统计
        if (success) {
            successfulReloads++;
            LOG_INFO("Successfully reloaded: %s", task.configName.c_str());
        } else {
            failedReloads++;
            LOG_ERROR("Failed to reload: %s", task.configName.c_str());
        }

        // 通知监听器
        for (auto& listener : listeners) {
            listener(task.configName, success);
        }

        // 记录重载历史
        RecordReloadHistory(task, success);
    }

    void RecordReloadHistory(const ReloadTask& task, bool success) {
        ReloadHistory record;
        record.configName = task.configName;
        record.filePath = task.filePath;
        record.reason = task.reason;
        record.timestamp = task.timestamp;
        record.success = success;
        record.duration = GetDuration(task.timestamp);

        // 保存到数据库或日志
        SaveReloadHistory(record);
    }
};
```

```cpp
// ConfigValidator.h
class ConfigValidator {
    struct ValidationRule {
        std::string fieldName;          // 字段名
        ValidationType type;            // 校验类型
        std::any parameters;             // 校验参数
        std::string errorMessage;        // 错误信息
    };

    struct ValidationSchema {
        std::string configName;          // 配置名
        std::vector<ValidationRule> rules;  // 校验规则
    };

    std::unordered_map<std::string, ValidationSchema> schemas;

public:
    enum ValidationType {
        REQUIRED,        // 必填
        TYPE_CHECK,      // 类型检查
        RANGE_CHECK,     // 范围检查
        REGEX_MATCH,     // 正则匹配
        CUSTOM_FUNC,     // 自定义函数
        REFERENCE_CHECK  // 引用检查
    };

    // 注册校验模式
    void RegisterSchema(const std::string& configName,
                       const ValidationSchema& schema) {
        schemas[configName] = schema;
    }

    // 校验配置
    ValidationResult Validate(const std::string& configName,
                             const void* configData) {
        ValidationResult result;
        result.configName = configName;
        result.isValid = true;

        auto it = schemas.find(configName);
        if (it == schemas.end()) {
            LOG_WARNING("No validation schema found for: %s", configName.c_str());
            return result;
        }

        const ValidationSchema& schema = it->second;

        for (const auto& rule : schema.rules) {
            if (!ValidateField(rule, configData, result)) {
                result.isValid = false;
                result.errors.push_back(rule.errorMessage);
            }
        }

        return result;
    }

    // 批量校验
    std::vector<ValidationResult> ValidateAll() {
        std::vector<ValidationResult> results;

        for (const auto& pair : schemas) {
            const std::string& configName = pair.first;
            auto configData = ConfigManager::Instance()->GetConfig(configName);

            if (configData) {
                auto result = Validate(configName, configData.get());
                results.push_back(result);
            }
        }

        return results;
    }

private:
    bool ValidateField(const ValidationRule& rule,
                      const void* configData,
                      ValidationResult& result) {
        switch (rule.type) {
            case REQUIRED:
                return ValidateRequired(rule, configData);
            case TYPE_CHECK:
                return ValidateType(rule, configData);
            case RANGE_CHECK:
                return ValidateRange(rule, configData);
            case REGEX_MATCH:
                return ValidateRegex(rule, configData);
            case CUSTOM_FUNC:
                return ValidateCustom(rule, configData);
            case REFERENCE_CHECK:
                return ValidateReference(rule, configData);
            default:
                return false;
        }
    }

    // 必填校验
    bool ValidateRequired(const ValidationRule& rule, const void* configData) {
        // 通过反射获取字段值
        auto value = GetFieldValue(configData, rule.fieldName);
        return value.has_value();
    }

    // 类型校验
    bool ValidateType(const ValidationRule& rule, const void* configData) {
        auto expectedType = std::any_cast<std::type_index>(rule.parameters);
        auto value = GetFieldValue(configData, rule.fieldName);

        if (!value.has_value()) return true;

        return value.type() == expectedType;
    }

    // 范围校验
    bool ValidateRange(const ValidationRule& rule, const void* configData) {
        auto range = std::any_cast<std::pair<double, double>>(rule.parameters);
        auto value = GetFieldValue(configData, rule.fieldName);

        if (!value.has_value()) return true;

        if (auto num = std::any_cast<double>(&value)) {
            return *num >= range.first && *num <= range.second;
        }

        return false;
    }

    // 引用校验
    bool ValidateReference(const ValidationRule& rule, const void* configData) {
        auto refConfig = std::any_cast<std::string>(rule.parameters);
        auto value = GetFieldValue(configData, rule.fieldName);

        if (!value.has_value()) return true;

        // 检查引用的配置是否存在
        auto refData = ConfigManager::Instance()->GetConfig(refConfig);
        return refData != nullptr;
    }
};
```

```cpp
// DependencyManager.h
class DependencyManager {
    struct DependencyNode {
        std::string configName;                // 配置名
        std::vector<std::string> dependencies;  // 依赖的配置
        std::vector<std::string> dependents;    // 被依赖的配置
        int inDegree;                          // 入度（用于拓扑排序）
    };

    std::unordered_map<std::string, DependencyNode> dependencyGraph;

public:
    // 添加依赖关系
    void AddDependency(const std::string& config,
                      const std::string& dependency) {
        auto& node = dependencyGraph[config];
        node.configName = config;
        node.dependencies.push_back(dependency);

        auto& depNode = dependencyGraph[dependency];
        depNode.configName = dependency;
        depNode.dependents.push_back(config);

        node.inDegree++;
    }

    // 获取拓扑排序
    std::vector<std::string> GetLoadOrder() {
        std::vector<std::string> result;
        std::queue<std::string> queue;
        std::unordered_map<std::string, int> tempInDegree;

        // 复制入度信息
        for (const auto& pair : dependencyGraph) {
            tempInDegree[pair.first] = pair.second.inDegree;
            if (pair.second.inDegree == 0) {
                queue.push(pair.first);
            }
        }

        // 拓扑排序
        while (!queue.empty()) {
            std::string current = queue.front();
            queue.pop();
            result.push_back(current);

            // 更新依赖此配置的其他配置
            const auto& node = dependencyGraph[current];
            for (const std::string& dependent : node.dependents) {
                if (--tempInDegree[dependent] == 0) {
                    queue.push(dependent);
                }
            }
        }

        // 检查是否存在循环依赖
        if (result.size() != dependencyGraph.size()) {
            LOG_ERROR("Circular dependency detected!");
            return {};
        }

        return result;
    }

    // 获取需要重新加载的配置
    std::vector<std::string> GetReloadChain(const std::string& config) {
        std::vector<std::string> chain;
        std::set<std::string> visited;

        GetReloadChainRecursive(config, chain, visited);
        return chain;
    }

    // 检查循环依赖
    bool HasCircularDependency() {
        std::set<std::string> visited;
        std::set<std::string> recursionStack;

        for (const auto& pair : dependencyGraph) {
            if (HasCircularDependencyRecursive(pair.first,
                                              visited,
                                              recursionStack)) {
                return true;
            }
        }

        return false;
    }

private:
    void GetReloadChainRecursive(const std::string& config,
                                std::vector<std::string>& chain,
                                std::set<std::string>& visited) {
        if (visited.find(config) != visited.end()) {
            return;
        }

        visited.insert(config);
        chain.push_back(config);

        // 添加所有依赖此配置的配置
        const auto& node = dependencyGraph[config];
        for (const std::string& dependent : node.dependents) {
            GetReloadChainRecursive(dependent, chain, visited);
        }
    }

    bool HasCircularDependencyRecursive(const std::string& config,
                                       std::set<std::string>& visited,
                                       std::set<std::string>& recursionStack) {
        visited.insert(config);
        recursionStack.insert(config);

        const auto& node = dependencyGraph[config];
        for (const std::string& dependency : node.dependencies) {
            if (recursionStack.find(dependency) != recursionStack.end()) {
                return true;  // 找到循环
            }

            if (visited.find(dependency) == visited.end() &&
                HasCircularDependencyRecursive(dependency,
                                               visited,
                                               recursionStack)) {
                return true;
            }
        }

        recursionStack.erase(config);
        return false;
    }
};
```

```cpp
class ConfigVersionManager {
```

### 7.1 高性能缓存实现

#### ConfigCache实现
```cpp
// ConfigCache.h
class ConfigCache {
    struct CacheEntry {
        void* data;
        std::type_index type;
        uint64_t lastAccess;
        uint32_t accessCount;
        uint32_t size;
        bool isPinned;  // 是否锁定（不被淘汰）
    };

    // LRU缓存实现
    class LRUCache {
        struct Node {
            std::string key;
            CacheEntry* entry;
            Node* prev;
            Node* next;

            Node(const std::string& k, CacheEntry* e)
                : key(k), entry(e), prev(nullptr), next(nullptr) {}
        };

        std::unordered_map<std::string, Node*> cache;
        Node* head;
        Node* tail;
        size_t capacity;
        size_t currentSize;
        std::mutex cacheMutex;

    public:
        LRUCache(size_t cap) : capacity(cap), currentSize(0) {
            head = new Node("", nullptr);
            tail = new Node("", nullptr);
            head->next = tail;
            tail->prev = head;
        }

        ~LRUCache() {
            for (auto& pair : cache) {
                delete pair.second;
            }
            delete head;
            delete tail;
        }

        CacheEntry* Get(const std::string& key) {
            std::lock_guard<std::mutex> lock(cacheMutex);
            auto it = cache.find(key);
            if (it == cache.end()) {
                return nullptr;
            }

            // 移到头部
            MoveToHead(it->second);
            return it->second->entry;
        }

        void Put(const std::string& key, CacheEntry* entry) {
            std::lock_guard<std::mutex> lock(cacheMutex);
            auto it = cache.find(key);

            if (it != cache.end()) {
                // 更新现有条目
                it->second->entry = entry;
                MoveToHead(it->second);
            } else {
                // 添加新条目
                Node* node = new Node(key, entry);
                cache[key] = node;
                AddToFront(node);
                currentSize += entry->size;

                // 检查容量
                while (currentSize > capacity) {
                    RemoveTail();
                }
            }
        }

        void Remove(const std::string& key) {
            std::lock_guard<std::mutex> lock(cacheMutex);
            auto it = cache.find(key);
            if (it != cache.end()) {
                RemoveNode(it->second);
                cache.erase(it);
            }
        }

    private:
        void MoveToHead(Node* node) {
            RemoveNode(node);
            AddToFront(node);
        }

        void AddToFront(Node* node) {
            node->prev = head;
            node->next = head->next;
            head->next->prev = node;
            head->next = node;
        }

        void RemoveNode(Node* node) {
            node->prev->next = node->next;
            node->next->prev = node->prev;
        }

        void RemoveTail() {
            Node* last = tail->prev;
            if (last != head) {
                RemoveNode(last);
                cache.erase(last->key);
                currentSize -= last->entry->size;
                delete last;
            }
        }
    };

    LRUCache lruCache;
    std::unordered_map<std::string, CacheEntry> entries;
    std::atomic<uint64_t> totalAccesses{0};
    std::atomic<uint64_t> cacheHits{0};

public:
    ConfigCache(size_t capacity) : lruCache(capacity) {}

    // 获取配置
    template<typename T>
    std::shared_ptr<T> Get(const std::string& key) {
        totalAccesses++;

        // 从LRU缓存获取
        CacheEntry* entry = lruCache.Get(key);
        if (entry) {
            cacheHits++;
            entry->lastAccess = GetCurrentTime();
            entry->accessCount++;

            if (entry->type == typeid(T)) {
                return std::static_pointer_cast<T>(
                    std::shared_ptr<void>(entry->data, [](void*){}));
            }
        }

        return nullptr;
    }

    // 存储配置
    template<typename T>
    void Put(const std::string& key, std::shared_ptr<T> config, bool pin = false) {
        CacheEntry entry;
        entry.data = config.get();
        entry.type = typeid(T);
        entry.lastAccess = GetCurrentTime();
        entry.accessCount = 1;
        entry.size = EstimateSize(config);
        entry.isPinned = pin;

        // 保存到条目映射
        entries[key] = entry;

        // 添加到LRU缓存
        lruCache.Put(key, &entries[key]);
    }

    // 预热缓存
    template<typename T>
    void Warmup(const std::string& key, std::function<std::shared_ptr<T>()> loader) {
        auto config = Get<T>(key);
        if (!config) {
            config = loader();
            if (config) {
                Put(key, config);
            }
        }
    }

    // 批量预热
    void BatchWarmup(const std::vector<std::string>& keys,
                    std::function<std::shared_ptr<void>(const std::string&)> loader) {
        std::vector<std::future<void>> futures;

        for (const auto& key : keys) {
            futures.push_back(std::async(std::launch::async, [this, &key, &loader]() {
                auto config = loader(key);
                if (config) {
                    // 根据实际类型存储
                    // 这里需要类型信息的支持
                }
            }));
        }

        // 等待所有预热完成
        for (auto& future : futures) {
            future.wait();
        }
    }

    // 获取缓存统计
    CacheStats GetStats() const {
        CacheStats stats;
        stats.totalAccesses = totalAccesses.load();
        stats.cacheHits = cacheHits.load();
        stats.hitRate = stats.totalAccesses > 0 ?
            (double)stats.cacheHits / stats.totalAccesses : 0.0;
        stats.totalEntries = entries.size();

        // 计算缓存大小
        for (const auto& pair : entries) {
            stats.totalSize += pair.second.size;
        }

        return stats;
    }

    // 清理过期缓存
    void Cleanup() {
        uint64_t now = GetCurrentTime();
        uint64_t expireTime = 30 * 60 * 1000;  // 30分钟过期

        for (auto it = entries.begin(); it != entries.end();) {
            if (!it->second.isPinned &&
                (now - it->second.lastAccess) > expireTime) {
                lruCache.Remove(it->first);
                it = entries.erase(it);
            } else {
                ++it;
            }
        }
    }

private:
```
