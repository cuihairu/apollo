## 1. 系统概述

### 1.1 设计目标
- **高性能**：支持数千玩家的实时属性同步
- **低延迟**：属性变化延迟控制在50ms内
- **一致性**：保证客户端与服务器属性一致
- **灵活性**：支持动态添加新属性类型
- **可扩展**：支持复杂的属性计算公式

### 2.2 属性定义管理器

#### AttributeDefManager实现
```cpp
// AttributeDefManager.h
class AttributeDefManager {
    struct AttributeDefinition {
        uint32_t id;
        std::string name;
        std::string type;
        AttributeGroup group;
        int64_t defaultValue;
        int64_t minValue;
        int64_t maxValue;
        ValueType valueType;
        uint32_t syncFlag;
        std::string formula;
        std::string description;
        bool isVisible;
        bool isSaveable;
        uint32_t cacheTime;

        // 运行时缓存
        std::unique_ptr<Formula> compiledFormula;
        uint64_t lastCacheUpdate;
        int64_t cachedValue;
    };

    std::unordered_map<uint32_t, AttributeDefinition> attributeDefs;
    std::unordered_map<std::string, uint32_t> nameToId;
    std::unordered_map<AttributeGroup, std::vector<uint32_t>> groupAttributes;

    // 属性计算缓存
    LRUCache<uint32_t, int64_t> calcCache{1000};

public:
    static AttributeDefManager* Instance() {
        static AttributeDefManager instance;
        return &instance;
    }

    // 初始化
    bool Initialize() {
        // 加载属性配置
        if (!LoadAttributeConfig()) {
            return false;
        }

        // 编译计算公式
        CompileFormulas();

        // 构建索引
        BuildIndexes();

        return true;
    }

    // 获取属性定义
    const AttributeDefinition* GetDefinition(uint32_t attrId) const {
        auto it = attributeDefs.find(attrId);
        return it != attributeDefs.end() ? &it->second : nullptr;
    }

    // 通过名称获取ID
    uint32_t GetAttributeId(const std::string& name) const {
        auto it = nameToId.find(name);
        return it != nameToId.end() ? it->second : 0;
    }

    // 获取组内所有属性
    const std::vector<uint32_t>& GetGroupAttributes(AttributeGroup group) const {
        static const std::vector<uint32_t> empty;
        auto it = groupAttributes.find(group);
        return it != groupAttributes.end() ? it->second : empty;
    }

    // 检查属性有效性
    bool IsValidAttribute(uint32_t attrId) const {
        return attributeDefs.find(attrId) != attributeDefs.end();
    }

    // 计算属性值
    int64_t CalculateAttribute(uint32_t attrId, Player* player) const {
        auto def = GetDefinition(attrId);
        if (!def || def->formula.empty()) {
            return def ? def->defaultValue : 0;
        }

        // 检查缓存
        uint64_t now = GetCurrentTime();
        if (def->cacheTime > 0 &&
            (now - def->lastCacheUpdate) < def->cacheTime) {
            return def->cachedValue;
        }

        // 执行计算
        int64_t result = 0;
        if (def->compiledFormula) {
            FormulaContext context;
            context.player = player;
            context.attributeId = attrId;

            result = def->compiledFormula->Evaluate(context);
        }

        // 应用数值范围限制
        result = std::max(def->minValue, std::min(def->maxValue, result));

        // 更新缓存
        def->cachedValue = result;
        def->lastCacheUpdate = now;

        return result;
    }

    // 批量计算属性
    void CalculateGroup(AttributeGroup group, Player* player,
                         std::map<uint32_t, int64_t>& results) const {
        const auto& attrIds = GetGroupAttributes(group);

        for (uint32_t attrId : attrIds) {
            int64_t value = CalculateAttribute(attrId, player);
            results[attrId] = value;
        }
    }

    // 获取同步标记
    uint32_t GetSyncFlag(uint32_t attrId) const {
        auto def = GetDefinition(attrId);
        return def ? def->syncFlag : 0;
    }

    // 检查是否需要保存
    bool IsSaveable(uint32_t attrId) const {
        auto def = GetDefinition(attrId);
        return def ? def->isSaveable : false;
    }

    // 检查是否为外观属性
    bool IsAppearance(uint32_t attrId) const {
        return (GetSyncFlag(attrId) & PROPERTY_APPR) != 0;
    }

private:
    // 加载属性配置
    bool LoadAttributeConfig() {
        auto config = ConfigManager::Instance()->GetConfig<AttributeConfig>(
            "AttributeConfig");
        if (!config) {
            LOG_ERROR("Failed to load AttributeConfig");
            return false;
        }

        for (const auto& cfg : config->data) {
            AttributeDefinition def;
            def.id = cfg.ID;
            def.name = cfg.Name;
            def.type = cfg.Type;
            def.group = StringToAttributeGroup(cfg.Group);
            def.defaultValue = cfg.Default;
            def.minValue = cfg.MinValue;
            def.maxValue = cfg.MaxValue;
            def.valueType = StringToValueType(cfg.ValueType);
            def.syncFlag = ParseSyncFlag(cfg.SyncFlag);
            def.formula = cfg.Formula;
            def.description = cfg.Description;
            def.isVisible = cfg.IsVisible;
            def.isSaveable = cfg.IsSaveable;
            def.cacheTime = cfg.CacheTime;
            def.lastCacheUpdate = 0;
            def.cachedValue = 0;

            attributeDefs[def.id] = def;
        }

        return true;
    }

    // 编译公式
    void CompileFormulas() {
        for (auto& pair : attributeDefs) {
            AttributeDefinition& def = pair.second;
            if (!def.formula.empty()) {
                try {
                    def.compiledFormula = std::make_unique<Formula>(def.formula);
                    LOG_DEBUG("Compiled formula for attribute %u", def.id);
                } catch (const std::exception& e) {
                    LOG_ERROR("Failed to compile formula for attribute %u: %s",
                             def.id, e.what());
                }
            }
        }
    }

    // 构建索引
    void BuildIndexes() {
        // 名称到ID的映射
        for (const auto& pair : attributeDefs) {
            nameToId[pair.second.name] = pair.first;
        }

        // 组索引
        groupAttributes.clear();
        for (const auto& pair : attributeDefs) {
            groupAttributes[pair.second.group].push_back(pair.first);
        }
    }
};
```

## 3. 属性存储结构

### 3.1 属性容器设计

#### AttributeContainer实现
```cpp
// AttributeContainer.h
class AttributeContainer {
private:
    // 属性值存储
    std::unordered_map<uint32_t, int64_t> attributes;
    std::unordered_map<uint32_t, int64_t> baseAttributes;  // 基础值
    std::unordered_map<uint32_t, std::vector<AttributeModifier>> modifiers;  // 修饰器

    // 变化追踪
    std::set<uint32_t> changedAttributes;
    std::unordered_map<uint32_t, AttributeChange> changeHistory;
    uint64_t lastSyncTime;

    // 互斥锁
    mutable std::shared_mutex mutex;

public:
    // 设置属性值
    void SetAttribute(uint32_t attrId, int64_t value) {
        std::unique_lock lock(mutex);

        auto& attr = attributes[attrId];
        if (attr != value) {
            attr = value;
            MarkChanged(attrId, PROPERTY_SET, value - attr);
        }
    }

    // 获取属性值
    int64_t GetAttribute(uint32_t attrId) const {
        std::shared_lock lock(mutex);

        // 检查是否有修饰器
        auto itMod = modifiers.find(attrId);
        if (itMod != modifiers.end() && !itMod->second.empty()) {
            // 计算最终值
            int64_t base = GetBaseAttribute(attrId);
            return CalculateFinalValue(attrId, base);
        }

        auto it = attributes.find(attrId);
        return it != attributes.end() ? it->second : GetDefaultValue(attrId);
    }

    // 设置基础值
    void SetBaseAttribute(uint32_t attrId, int64_t value) {
        std::unique_lock lock(mutex);

        baseAttributes[attrId] = value;
        RecalculateAttribute(attrId);
    }

    // 获取基础值
    int64_t GetBaseAttribute(uint32_t attrId) const {
        std::shared_lock lock(mutex);

        auto it = baseAttributes.find(attrId);
        if (it != baseAttributes.end()) {
            return it->second;
        }

        // 如果没有基础值，从当前值获取
        auto itAttr = attributes.find(attrId);
        return itAttr != attributes.end() ? itAttr->second : GetDefaultValue(attrId);
    }

    // 添加修饰器
    void AddModifier(uint32_t attrId, const AttributeModifier& modifier) {
        std::unique_lock lock(mutex);

        auto& list = modifiers[attrId];

        // 检查是否已存在相同来源的修饰器
        for (auto& mod : list) {
            if (mod.source == modifier.source && mod.type == modifier.type) {
                mod.value = modifier.value;  // 更新值
                RecalculateAttribute(attrId);
                return;
            }
        }

        // 添加新修饰器
        list.push_back(modifier);
        RecalculateAttribute(attrId);
    }

    // 移除修饰器
    void RemoveModifier(uint32_t attrId, const std::string& source) {
        std::unique_lock lock(mutex);

        auto it = modifiers.find(attrId);
        if (it != modifiers.end()) {
            auto& list = it->second;
            list.erase(std::remove_if(list.begin(), list.end(),
                [&source](const AttributeModifier& mod) {
                    return mod.source == source;
                }), list.end());

            RecalculateAttribute(attrId);
        }
    }

    // 批量设置属性
    void SetAttributes(const std::map<uint32_t, int64_t>& attrs) {
        std::unique_lock lock(mutex);

        for (const auto& pair : attrs) {
            auto& attr = attributes[pair.first];
            if (attr != pair.second) {
                attr = pair.second;
                MarkChanged(pair.first, PROPERTY_SET, pair.second - attr);
            }
        }
    }

    // 获取所有属性
    std::map<uint32_t, int64_t> GetAllAttributes() const {
        std::shared_lock lock(mutex);
        return attributes;
    }

    // 获取变化的属性
    std::vector<AttributeChange> GetChanges() {
        std::unique_lock lock(mutex);

        std::vector<AttributeChange> changes;

        for (uint32_t attrId : changedAttributes) {
            auto it = changeHistory.find(attrId);
            if (it != changeHistory.end()) {
                changes.push_back(it->second);
            }
        }

        return changes;
    }

    // 清除变化标记
    void ClearChanges() {
        std::unique_lock lock(mutex);
        changedAttributes.clear();
        changeHistory.clear();
        lastSyncTime = GetCurrentTime();
    }

    // 获取指定组的属性
    std::map<uint32_t, int64_t> GetGroupAttributes(AttributeGroup group) const {
        std::shared_lock lock(mutex);

        std::map<uint32_t, int64_t> result;

        const auto& attrIds = AttributeDefManager::Instance()->GetGroupAttributes(group);
        for (uint32_t attrId : attrIds) {
            auto it = attributes.find(attrId);
            if (it != attributes.end()) {
                result[attrId] = it->second;
            }
        }

        return result;
    }

    // 序列化属性
    void Serialize(Json::Value& json) const {
        std::shared_lock lock(mutex);

        for (const auto& pair : attributes) {
            uint32_t attrId = pair.first;
            int64_t value = pair.second;

            auto def = AttributeDefManager::Instance()->GetDefinition(attrId);
            if (def && def->isSaveable) {
                json[std::to_string(attrId)] = static_cast<Json::Int64>(value);
            }
        }
    }

    // 反序列化属性
    void Deserialize(const Json::Value& json) {
        std::unique_lock lock(mutex);

        for (const auto& key : json.getMemberNames()) {
            uint32_t attrId = std::stoul(key);
            int64_t value = json[key].asInt64();

            auto def = AttributeDefManager::Instance()->GetDefinition(attrId);
            if (def && def->isSaveable) {
                attributes[attrId] = value;
            }
        }
    }

private:
    // 标记属性变化
    void MarkChanged(uint32_t attrId, uint32_t changeType, int64_t delta) {
        changedAttributes.insert(attrId);

        AttributeChange change;
        change.attributeId = attrId;
        change.oldValue = attributes[attrId] - delta;
        change.newValue = attributes[attrId];
        change.delta = delta;
        change.changeType = changeType;
        change.timestamp = GetCurrentTime();

        changeHistory[attrId] = change;
    }

    // 重新计算属性
    void RecalculateAttribute(uint32_t attrId) {
        int64_t base = GetBaseAttribute(attrId);
        int64_t finalValue = CalculateFinalValue(attrId, base);

        auto& current = attributes[attrId];
        if (current != finalValue) {
            MarkChanged(attrId, PROPERTY_RECALC, finalValue - current);
            current = finalValue;
        }
    }

    // 计算最终值
    int64_t CalculateFinalValue(uint32_t attrId, int64_t base) const {
        auto it = modifiers.find(attrId);
        if (it == modifiers.end() || it->second.empty()) {
            return base;
        }

        int64_t result = base;
        int64_t addValue = 0;
        float mulValue = 1.0f;

        // 按类型处理修饰器
        for (const auto& mod : it->second) {
            switch (mod.type) {
                case MODIFIER_ADD:
                    addValue += mod.value;
                    break;
                case MODIFIER_SUB:
                    addValue -= mod.value;
                    break;
                case MODIFIER_MUL:
                    mulValue *= (1.0f + mod.value / 100.0f);
                    break;
                case MODIFIER_FINAL_ADD:
                    result += mod.value;
                    break;
            }
        }

        // 应用加法和乘法
        result = static_cast<int64_t>((result + addValue) * mulValue);

        return result;
    }

    // 获取默认值
    int64_t GetDefaultValue(uint32_t attrId) const {
        auto def = AttributeDefManager::Instance()->GetDefinition(attrId);
        return def ? def->defaultValue : 0;
    }
};
```

```cpp
// AttributeModifier.h
class AttributeModifier {
public:
    enum ModifierType {
        MODIFIER_ADD = 1,      // 加法修饰器
        MODIFIER_SUB = 2,      // 减法修饰器
        MODIFIER_MUL = 3,      // 乘法修饰器（百分比）
        MODIFIER_FINAL_ADD = 4, // 最终加法
        MODIFIER_OVERRIDE = 5,  // 覆盖值
    };

    enum SourceType {
        SOURCE_EQUIP = 1,      // 装备
        SOURCE_BUFF = 2,       // Buff
        SOURCE_SKILL = 3,      // 技能
        SOURCE_PET = 4,        // 宠物
        SOURCE_GUILD = 5,      // 公会
        SOURCE_TITLE = 6,      // 称号
        SOURCE_ENVIRONMENT = 7, // 环境
        SOURCE_GM = 8,         // GM命令
    };

    ModifierType type;         // 修饰器类型
    int64_t value;            // 修饰器值
    std::string source;       // 来源标识
    SourceType sourceType;    // 来源类型
    uint64_t expireTime;      // 过期时间（0表示永久）
    bool isStackable;        // 是否可叠加
    int maxStack;             // 最大叠加数

    AttributeModifier(ModifierType t = MODIFIER_ADD,
                     int64_t v = 0,
                     const std::string& src = "",
                     SourceType srcType = SOURCE_BUFF)
```

```cpp
// AttributeSyncManager.h
class AttributeSyncManager {
    struct SyncTask {
        uint32_t playerId;
        uint32_t attributeId;
        int64_t oldValue;
        int64_t newValue;
        uint32_t syncFlag;
        uint64_t sequence;
        uint64_t timestamp;
        std::set<uint32_t> excludeList;  // 排除的玩家
    };

    // 同步队列
    ThreadSafeQueue<SyncTask> syncQueue;
    std::thread syncThread;
    std::atomic<bool> running{true};

    // 批量同步缓冲
    std::unordered_map<uint32_t, std::vector<SyncTask>> playerSyncBuffer;
    std::mutex bufferMutex;
    uint64_t lastFlushTime;
    static const uint64_t BATCH_INTERVAL = 50; // 50ms批处理间隔

    // 序列号生成器
    std::atomic<uint64_t> sequenceGenerator{1};

public:
    AttributeSyncManager() : lastFlushTime(0) {
        syncThread = std::thread(&AttributeSyncManager::SyncLoop, this);
    }

    ~AttributeSyncManager() {
        running = false;
        if (syncThread.joinable()) {
            syncThread.join();
        }
    }

    // 同步属性变化
    void SyncAttributeChange(uint32_t playerId,
                              uint32_t attributeId,
                              int64_t oldValue,
                              int64_t newValue) {
        uint32_t syncFlag = AttributeDefManager::Instance()->GetSyncFlag(attributeId);
        if (syncFlag == 0) {
            return;  // 不需要同步
        }

        SyncTask task;
        task.playerId = playerId;
        task.attributeId = attributeId;
        task.oldValue = oldValue;
        task.newValue = newValue;
        task.syncFlag = syncFlag;
        task.sequence = sequenceGenerator++;
        task.timestamp = GetCurrentTime();

        // 根据同步标记决定处理方式
        if (syncFlag & PROPERTY_SELF) {
            SendToSelf(task);
        }

        if (syncFlag & PROPERTY_APPR) {
            // 外观变化需要立即同步
            if (IsAppearanceAttribute(attributeId)) {
                SendImmediateSync(task);
            } else {
                AddToBatch(task);
            }
        }

        if (syncFlag & PROPERTY_PRO) {
            AddToBatch(task);
        }

        if (syncFlag & PROPERTY_TEAM) {
            SendToTeam(task);
        }

        if (syncFlag & PROPERTY_GUILD) {
            SendToGuild(task);
        }

        if (syncFlag & PROPERTY_WORLD) {
            SendToWorld(task);
        }
    }

    // 立即同步
    void SyncImmediately(uint32_t playerId, uint32_t attributeId) {
        Player* player = PlayerManager::Instance()->GetPlayer(playerId);
        if (!player) return;

        int64_t value = player->GetAttribute(attributeId);
        uint32_t syncFlag = AttributeDefManager::Instance()->GetSyncFlag(attributeId);

        SyncTask task;
        task.playerId = playerId;
        task.attributeId = attributeId;
        task.oldValue = 0;
        task.newValue = value;
        task.syncFlag = syncFlag;
        task.sequence = sequenceGenerator++;
        task.timestamp = GetCurrentTime();

        SendImmediateSync(task);
    }

    // 强制全量同步
    void ForceFullSync(uint32_t playerId) {
        Player* player = PlayerManager::Instance()->GetPlayer(playerId);
        if (!player) return;

        // 获取所有需要同步的属性
        auto allAttrs = player->GetAttributeContainer()->GetAllAttributes();

        AttributeUpdateBatch batch;
        batch.playerId = playerId;
        batch.timestamp = GetCurrentTime();
        batch.isFullSync = true;

        for (const auto& pair : allAttrs) {
            uint32_t attrId = pair.first;
            int64_t value = pair.second;
            uint32_t syncFlag = AttributeDefManager::Instance()->GetSyncFlag(attrId);

            if (syncFlag != 0) {
                AttributeData data;
                data.attributeId = attrId;
                data.value = value;
                data.syncFlag = syncFlag;
                batch.attributes.push_back(data);
            }
        }

        // 发送批量更新
        SendAttributeUpdateBatch(batch);
    }

    // 获取同步统计
    SyncStats GetStats() const {
        return syncStats;
    }

private:
    SyncStats syncStats;

    // 同步循环
    void SyncLoop() {
        while (running) {
            uint64_t now = GetCurrentTime();

            // 处理队列中的同步任务
            std::vector<SyncTask> tasks;
            syncQueue.try_pop_bulk(tasks, 100);

            for (const auto& task : tasks) {
                ProcessSyncTask(task);
            }

            // 定期刷新批处理缓冲
            if (now - lastFlushTime > BATCH_INTERVAL) {
                FlushBatchBuffer();
                lastFlushTime = now;
            }

            // 短暂休眠
            std::this_thread::sleep_for(std::chrono::milliseconds(10));
        }
    }

    // 处理同步任务
    void ProcessSyncTask(const SyncTask& task) {
        uint32_t syncFlag = task.syncFlag;

        // 发送给自己
        if (syncFlag & PROPERTY_SELF) {
            SendToSelf(task);
        }

        // 根据类型发送给其他玩家
        if (syncFlag & PROPERTY_APPR) {
            SendToAOI(task);
        }

        if (syncFlag & PROPERTY_PRO) {
            SendToAOI(task);
        }
    }

    // 发送给自己
    void SendToSelf(const SyncTask& task) {
        Player* player = PlayerManager::Instance()->GetPlayer(task.playerId);
        if (!player) return;

        AttributeUpdate msg;
        msg.player_id = task.playerId;
        msg.add_attributes()->set_attribute_id(task.attributeId);
        msg.mutable_attributes(0)->set_value(task.newValue);
        msg.mutable_attributes(0)->set_sequence(task.sequence);
        msg.mutable_attributes(0)->set_timestamp(task.timestamp);

        player->SendPacket(CMD_ATTRIBUTE_UPDATE, msg);
        syncStats.selfSyncs++;
    }

    // 发送给AOI内的玩家
    void SendToAOI(const SyncTask& task) {
        Player* player = PlayerManager::Instance()->GetPlayer(task.playerId);
        if (!player) return;

        // 获取视野内的玩家
        auto aoiPlayers = AOIManager::Instance()->GetPlayersInRange(
            player->GetPosition(), player->GetViewRadius());

        // 构建消息
        AttributeUpdate msg;
        msg.set_player_id(task.playerId);
        auto* attr = msg.add_attributes();
        attr->set_attribute_id(task.attributeId);
        attr->set_value(task.newValue);
        attr->set_sequence(task.sequence);
        attr->set_timestamp(task.timestamp);

        // 广播给AOI内的玩家（排除自己）
        for (uint32_t watcherId : aoiPlayers) {
            if (watcherId == task.playerId) continue;
            if (task.excludeList.find(watcherId) != task.excludeList.end()) continue;

            Player* watcher = PlayerManager::Instance()->GetPlayer(watcherId);
            if (watcher) {
                watcher->SendPacket(CMD_OTHER_ATTRIBUTE_UPDATE, msg);
                syncStats.aoiSyncs++;
            }
        }
    }

    // 发送给队伍
    void SendToTeam(const SyncTask& task) {
        Player* player = PlayerManager::Instance()->GetPlayer(task.playerId);
        if (!player || !player->GetTeamId()) return;

        Team* team = TeamManager::Instance()->GetTeam(player->GetTeamId());
        if (!team) return;

        AttributeUpdate msg;
        msg.set_player_id(task.playerId);
        auto* attr = msg.add_attributes();
        attr->set_attribute_id(task.attributeId);
        attr->set_value(task.newValue);
        attr->set_sequence(task.sequence);
        attr->set_timestamp(task.timestamp);

        for (uint32_t memberId : team->GetMembers()) {
            if (memberId == task.playerId) continue;

            Player* member = PlayerManager::Instance()->GetPlayer(memberId);
            if (member) {
                member->SendPacket(CMD_TEAM_ATTRIBUTE_UPDATE, msg);
                syncStats.teamSyncs++;
            }
        }
    }

    // 发送给公会
    void SendToGuild(const SyncTask& task) {
        Player* player = PlayerManager::Instance()->GetPlayer(task.playerId);
        if (!player || !player->GetGuildId()) return;

        Guild* guild = GuildManager::Instance()->GetGuild(player->GetGuildId());
        if (!guild) return;

        // 只同步特定的公会属性（如等级、称号等）
        if (!IsGuildAttribute(task.attributeId)) {
            return;
        }

        AttributeUpdate msg;
        msg.set_player_id(task.playerId);
        auto* attr = msg.add_attributes();
        attr->set_attribute_id(task.attributeId);
        attr->set_value(task.newValue);
        attr->set_sequence(task.sequence);
        attr->set_timestamp(task.timestamp);

        for (uint32_t memberId : guild->GetOnlineMembers()) {
            if (memberId == task.playerId) continue;

            Player* member = PlayerManager::Instance()->GetPlayer(memberId);
            if (member) {
                member->SendPacket(CMD_GUILD_ATTRIBUTE_UPDATE, msg);
                syncStats.guildSyncs++;
            }
        }
    }

    // 添加到批处理缓冲
    void AddToBatch(const SyncTask& task) {
        std::lock_guard<std::mutex> lock(bufferMutex);
        playerSyncBuffer[task.playerId].push_back(task);
    }

    // 刷新批处理缓冲
    void FlushBatchBuffer() {
        std::lock_guard<std::mutex> lock(bufferMutex);

        for (auto& pair : playerSyncBuffer) {
            uint32_t playerId = pair.first;
            auto& tasks = pair.second;

            if (tasks.empty()) continue;

            // 构建批量更新消息
            AttributeUpdateBatch batch;
            batch.playerId = playerId;
            batch.timestamp = GetCurrentTime();
            batch.isFullSync = false;

            for (const auto& task : tasks) {
                AttributeData data;
                data.attributeId = task.attributeId;
                data.value = task.newValue;
                data.syncFlag = task.syncFlag;
                data.sequence = task.sequence;
                batch.attributes.push_back(data);
            }

            // 发送批量更新
            SendAttributeUpdateBatch(batch);

            // 清空缓冲
            tasks.clear();
        }
    }

    // 发送属性批量更新
    void SendAttributeUpdateBatch(const AttributeUpdateBatch& batch) {
        Player* player = PlayerManager::Instance()->GetPlayer(batch.playerId);
        if (!player) return;

        // 构建Protobuf消息
        gameMsg::AttributeUpdateBatch msg;
        msg.set_player_id(batch.playerId);
        msg.set_is_full_sync(batch.isFullSync);
        msg.set_timestamp(batch.timestamp);

        for (const auto& attr : batch.attributes) {
            auto* attrMsg = msg.add_attributes();
            attrMsg->set_attribute_id(attr.attributeId);
            attrMsg->set_value(attr.value);
            attrMsg->set_sequence(attr.sequence);
        }

        // 发送给玩家自己
        if (batch.attributes.empty() ||
            (batch.attributes[0].syncFlag & PROPERTY_SELF)) {
            player->SendPacket(CMD_ATTRIBUTE_UPDATE_BATCH, msg);
        }

        // 发送给AOI内的其他玩家
        if (batch.attributes.empty() ||
            (batch.attributes[0].syncFlag & (PROPERTY_APPR | PROPERTY_PRO))) {
            BroadcastToAOI(player, msg);
        }

        syncStats.batchSyncs++;
    }

    // 广播到AOI
    void BroadcastToAOI(Player* player, const gameMsg::AttributeUpdateBatch& msg) {
        auto aoiPlayers = AOIManager::Instance()->GetPlayersInRange(
            player->GetPosition(), player->GetViewRadius());

        for (uint32_t watcherId : aoiPlayers) {
            if (watcherId == player->GetId()) continue;

            Player* watcher = PlayerManager::Instance()->GetPlayer(watcherId);
            if (watcher) {
                watcher->SendPacket(CMD_OTHER_ATTRIBUTE_BATCH_UPDATE, msg);
            }
        }
    }

    // 检查是否为外观属性
    bool IsAppearanceAttribute(uint32_t attrId) const {
        auto def = AttributeDefManager::Instance()->GetDefinition(attrId);
        return def ? (def->group == GROUP_APPEARANCE) : false;
    }

    // 检查是否为公会属性
    bool IsGuildAttribute(uint32_t attrId) const {
        auto def = AttributeDefManager::Instance()->GetDefinition(attrId);
        return def ? (def->syncFlag & PROPERTY_GUILD) != 0 : false;
    }
};
```

```cpp
// DifferentialSync.h
class DifferentialSync {
    struct AttributeSnapshot {
        uint64_t timestamp;
        std::unordered_map<uint32_t, int64_t> values;
        uint64_t checksum;
    };

    // 快照存储
    std::unordered_map<uint32_t, AttributeSnapshot> playerSnapshots;
    std::mutex snapshotMutex;

    // 变化阈值（小于此值的变化不被同步）
    std::unordered_map<uint32_t, int64_t> changeThresholds;

public:
    // 设置变化阈值
    void SetChangeThreshold(uint32_t attributeId, int64_t threshold) {
        changeThresholds[attributeId] = threshold;
    }

    // 创建快照
    void CreateSnapshot(uint32_t playerId,
                        const std::map<uint32_t, int64_t>& attributes) {
        std::lock_guard<std::mutex> lock(snapshotMutex);

        AttributeSnapshot& snapshot = playerSnapshots[playerId];
        snapshot.timestamp = GetCurrentTime();
        snapshot.values.clear();

        // 只保存需要同步的属性
        for (const auto& pair : attributes) {
            uint32_t attrId = pair.first;
            uint32_t syncFlag = AttributeDefManager::Instance()->GetSyncFlag(attrId);

            if (syncFlag != 0) {
                snapshot.values[attrId] = pair.second;
            }
        }

        // 计算校验和
        snapshot.checksum = CalculateChecksum(snapshot.values);

        LOG_DEBUG("Created snapshot for player %u with %lu attributes",
                  playerId, snapshot.values.size());
    }

    // 获取差分变化
    std::vector<AttributeChange> GetDifferentialChanges(uint32_t playerId,
                                                        const std::map<uint32_t, int64_t>& current) {
        std::lock_guard<std::mutex> lock(snapshotMutex);

        std::vector<AttributeChange> changes;
        auto it = playerSnapshots.find(playerId);

        if (it == playerSnapshots.end()) {
            // 没有快照，返回所有属性作为变化
            for (const auto& pair : current) {
                AttributeChange change;
                change.attributeId = pair.first;
                change.oldValue = 0;
                change.newValue = pair.second;
                change.delta = pair.second;
                changes.push_back(change);
            }
            return changes;
        }

        const AttributeSnapshot& snapshot = it->second;

        // 比较每个属性
        for (const auto& pair : current) {
            uint32_t attrId = pair.first;
            int64_t newValue = pair.second;

            auto itSnapshot = snapshot.values.find(attrId);
            int64_t oldValue = (itSnapshot != snapshot.values.end()) ?
                               itSnapshot->second : 0;

            // 检查变化是否超过阈值
            int64_t delta = newValue - oldValue;
            if (ShouldSyncChange(attrId, delta)) {
                AttributeChange change;
                change.attributeId = attrId;
                change.oldValue = oldValue;
                change.newValue = newValue;
                change.delta = delta;
                change.timestamp = GetCurrentTime();
                changes.push_back(change);
            }
        }

        return changes;
    }

    // 检查是否需要同步变化
    bool ShouldSyncChange(uint32_t attrId, int64_t delta) {
        if (delta == 0) return false;

        // 检查阈值
        auto it = changeThresholds.find(attrId);
        if (it != changeThresholds.end()) {
            return std::abs(delta) >= it->second;
        }

        // 默认阈值
        switch (GetAttributeGroup(attrId)) {
            case GROUP_BASIC:
                // 基础属性变化1就同步
                return std::abs(delta) >= 1;
            case GROUP_COMBAT:
                // 战斗属性变化0.1就同步
                return std::abs(delta) >= 1;
            case GROUP_APPEARANCE:
                // 外观属性任何变化都同步
                return true;
            default:
                return std::abs(delta) >= 1;
        }
    }

    // 更新快照
    void UpdateSnapshot(uint32_t playerId,
                         const std::map<uint32_t, int64_t>& attributes,
                         const std::vector<AttributeChange>& changes) {
        std::lock_guard<std::mutex> lock(snapshotMutex);

        auto it = playerSnapshots.find(playerId);
        if (it == playerSnapshots.end()) {
            return;
        }

        AttributeSnapshot& snapshot = it->second;

        // 只更新变化的属性
        for (const auto& change : changes) {
            snapshot.values[change.attributeId] = change.newValue;
        }

        // 更新时间戳
        snapshot.timestamp = GetCurrentTime();

        // 重新计算校验和
        snapshot.checksum = CalculateChecksum(snapshot.values);
    }

    // 检查快照完整性
    bool VerifySnapshotIntegrity(uint32_t playerId,
                                 const std::map<uint32_t, int64_t>& current) {
        std::lock_guard<std::mutex> lock(snapshotMutex);

        auto it = playerSnapshots.find(playerId);
        if (it == playerSnapshots.end()) {
            return false;
        }

        // 计算当前值的校验和
        uint64_t currentChecksum = CalculateChecksum(current);

        return currentChecksum == it->second.checksum;
    }

    // 清理过期快照
    void CleanupExpiredSnapshots(uint64_t expireTime = 300000) {  // 5分钟
        std::lock_guard<std::mutex> lock(snapshotMutex);

        uint64_t now = GetCurrentTime();
        auto it = playerSnapshots.begin();

        while (it != playerSnapshots.end()) {
            if (now - it->second.timestamp > expireTime) {
                LOG_DEBUG("Removing expired snapshot for player %u", it->first);
                it = playerSnapshots.erase(it);
            } else {
                ++it;
            }
        }
    }

private:
    // 计算校验和
    uint64_t CalculateChecksum(const std::map<uint32_t, int64_t>& values) {
        uint64_t checksum = 0;

        for (const auto& pair : values) {
            checksum ^= (static_cast<uint64_t>(pair.first) << 32) |
                         static_cast<uint64_t>(pair.second);
        }

        return checksum;
    }

    // 获取属性组
    AttributeGroup GetAttributeGroup(uint32_t attrId) {
        auto def = AttributeDefManager::Instance()->GetDefinition(attrId);
        return def ? def->group : GROUP_BASIC;
    }
};
```

```cpp
// AttributeTracker.h
class AttributeTracker {
    struct ChangeRecord {
        uint32_t attributeId;
        int64_t oldValue;
        int64_t newValue;
        int64_t delta;
        uint32_t changeType;
        uint64_t timestamp;
        std::string source;
        std::string reason;
    };

    // 变化记录存储
    std::unordered_map<uint32_t, std::deque<ChangeRecord>> playerChanges;
    std::mutex changesMutex;

    // 追踪配置
    size_t maxRecordsPerPlayer = 1000;
    uint64_t maxRecordAge = 24 * 60 * 60 * 1000;  // 24小时

public:
    // 记录属性变化
    void RecordChange(uint32_t playerId,
                      uint32_t attributeId,
                      int64_t oldValue,
                      int64_t newValue,
                      uint32_t changeType = PROPERTY_SET,
                      const std::string& source = "",
                      const std::string& reason = "") {
        std::lock_guard<std::mutex> lock(changesMutex);

        ChangeRecord record;
        record.attributeId = attributeId;
        record.oldValue = oldValue;
        record.newValue = newValue;
        record.delta = newValue - oldValue;
        record.changeType = changeType;
        record.timestamp = GetCurrentTime();
        record.source = source;
        record.reason = reason;

        auto& changes = playerChanges[playerId];
        changes.push_back(record);

        // 限制记录数量
        while (changes.size() > maxRecordsPerPlayer) {
            changes.pop_front();
        }

        LOG_DEBUG("Recorded attribute change for player %u: attr=%u, %lld->%lld",
                  playerId, attributeId, oldValue, newValue);
    }

    // 获取变化历史
    std::vector<ChangeRecord> GetChangeHistory(uint32_t playerId,
                                              uint32_t attributeId = 0,
                                              uint64_t startTime = 0,
                                              uint64_t endTime = 0) {
        std::lock_guard<std::mutex> lock(changesMutex);

        std::vector<ChangeRecord> result;
        auto it = playerChanges.find(playerId);

        if (it == playerChanges.end()) {
            return result;
        }

        const auto& changes = it->second;
        for (const auto& record : changes) {
            // 筛选属性ID
            if (attributeId > 0 && record.attributeId != attributeId) {
                continue;
            }

            // 筛选时间范围
            if (startTime > 0 && record.timestamp < startTime) {
                continue;
            }
            if (endTime > 0 && record.timestamp > endTime) {
                continue;
            }

            result.push_back(record);
        }

        return result;
    }

    // 生成变化报告
    ChangeReport GenerateReport(uint32_t playerId,
                               uint64_t duration = 60000) {  // 默认1分钟
        uint64_t endTime = GetCurrentTime();
        uint64_t startTime = endTime - duration;

        ChangeReport report;
        report.playerId = playerId;
        report.startTime = startTime;
        report.endTime = endTime;
        report.totalChanges = 0;

        std::map<uint32_t, AttributeSummary> summaries;

        auto changes = GetChangeHistory(playerId, 0, startTime, endTime);
        for (const auto& record : changes) {
            report.totalChanges++;

            auto& summary = summaries[record.attributeId];
            summary.attributeId = record.attributeId;
            summary.changeCount++;
            summary.totalDelta += record.delta;

            if (record.delta > 0) {
                summary.increaseCount++;
                summary.totalIncrease += record.delta;
            } else if (record.delta < 0) {
                summary.decreaseCount++;
                summary.totalDecrease += record.delta;
            }

            // 记录最大值和最小值
            if (summary.maxValue == 0 || record.newValue > summary.maxValue) {
                summary.maxValue = record.newValue;
                summary.maxTime = record.timestamp;
            }
            if (summary.minValue == 0 || record.newValue < summary.minValue) {
                summary.minValue = record.newValue;
                summary.minTime = record.timestamp;
            }

            // 统计变化来源
            summary.sources[record.source]++;
        }

        // 转换为报告格式
        for (const auto& pair : summaries) {
            report.attributeSummaries.push_back(pair.second);
        }

        // 排序（按变化次数）
        std::sort(report.attributeSummaries.begin(), report.attributeSummaries.end(),
            [](const AttributeSummary& a, const AttributeSummary& b) {
                return a.changeCount > b.changeCount;
            });

        return report;
    }

    // 检测异常变化
    std::vector<AnomalyChange> DetectAnomalies(uint32_t playerId,
                                                uint64_t duration = 300000) {  // 5分钟
        uint64_t endTime = GetCurrentTime();
        uint64_t startTime = endTime - duration;

        std::vector<AnomalyChange> anomalies;
        auto changes = GetChangeHistory(playerId, 0, startTime, endTime);

        // 按属性分组
        std::map<uint32_t, std::vector<ChangeRecord>> attrChanges;
        for (const auto& record : changes) {
            attrChanges[record.attributeId].push_back(record);
        }

        // 检测每种属性的异常
        for (const auto& pair : attrChanges) {
            uint32_t attrId = pair.first;
            const auto& attrRecords = pair.second;

            // 检测高频变化
            if (attrRecords.size() > 100) {  // 5分钟内变化超过100次
                AnomalyChange anomaly;
                anomaly.attributeId = attrId;
                anomaly.type = ANOMALY_HIGH_FREQUENCY;
                anomaly.description = "属性变化过于频繁";
                anomaly.count = attrRecords.size();
                anomalies.push_back(anomaly);
            }

            // 检测大值变化
            for (const auto& record : attrRecords) {
                if (std::abs(record.delta) > GetAnomalyThreshold(attrId)) {
                    AnomalyChange anomaly;
                    anomaly.attributeId = attrId;
                    anomaly.type = ANOMALY_LARGE_DELTA;
                    anomaly.description = "属性变化值过大";
                    anomaly.delta = record.delta;
                    anomaly.timestamp = record.timestamp;
                    anomalies.push_back(anomaly);
                }
            }

            // 检测异常模式（如连续增长）
            if (attrRecords.size() >= 10) {
                bool continuousIncrease = true;
                for (size_t i = 1; i < attrRecords.size(); ++i) {
                    if (attrRecords[i].delta <= 0) {
                        continuousIncrease = false;
                        break;
                    }
                }

                if (continuousIncrease) {
                    AnomalyChange anomaly;
                    anomaly.attributeId = attrId;
                    anomaly.type = ANOMALY_CONTINUOUS_INCREASE;
                    anomaly.description = "属性持续增长";
                    anomaly.count = attrRecords.size();
                    anomalies.push_back(anomaly);
                }
            }
        }

        return anomalies;
    }

    // 清理过期记录
    void CleanupExpiredRecords() {
        std::lock_guard<std::mutex> lock(changesMutex);

        uint64_t now = GetCurrentTime();
        auto it = playerChanges.begin();

        while (it != playerChanges.end()) {
            auto& changes = it->second;

            // 移除过期记录
            while (!changes.empty() &&
                   (now - changes.front().timestamp) > maxRecordAge) {
                changes.pop_front();
            }

            // 如果没有记录了，移除整个条目
            if (changes.empty()) {
                it = playerChanges.erase(it);
            } else {
                ++it;
            }
        }
    }

    // 导出变化日志
    void ExportLogs(uint32_t playerId, const std::string& filePath) {
        std::lock_guard<std::mutex> lock(changesMutex);

        auto it = playerChanges.find(playerId);
        if (it == playerChanges.end()) {
            return;
        }

        std::ofstream file(filePath);
        if (!file.is_open()) {
            LOG_ERROR("Failed to open file for export: %s", filePath.c_str());
            return;
        }

        file << "Timestamp,AttributeID,OldValue,NewValue,Delta,Type,Source,Reason\n";

        for (const auto& record : it->second) {
            file << record.timestamp << ","
                 << record.attributeId << ","
                 << record.oldValue << ","
                 << record.newValue << ","
                 << record.delta << ","
                 << record.changeType << ","
                 << record.source << ","
                 << record.reason << "\n";
        }

        file.close();
        LOG_INFO("Exported %zu change records for player %u to %s",
                 it->second.size(), playerId, filePath.c_str());
    }

private:
    // 获取异常阈值
    int64_t GetAnomalyThreshold(uint32_t attrId) {
        auto def = AttributeDefManager::Instance()->GetDefinition(attrId);
        if (!def) return 1000;

        // 根据属性类型设置阈值
        switch (def->group) {
            case GROUP_BASIC:
                return def->maxValue / 10;  // 最大值的10%
            case GROUP_COMBAT:
                return 10000;  // 1万点
            case GROUP_STATUS:
                return 1000000;  // 100万经验
            case GROUP_APPEARANCE:
                return 0;  // 任何外观变化都算异常
            default:
                return 1000;
        }
    }
```

```cpp
class AttributePersistence {
    struct PersistenceTask {

                                const std::vector<AttributeData>& attributes,
                                uint32_t priority = 5) {
        if (attributes.empty()) return;

        PersistenceTask task;
        task.playerId = playerId;
        task.type = PERSIST_BATCH;
        task.timestamp = GetCurrentTime();
        task.priority = priority;
        task.attributes = attributes;

        // 检查是否包含关键属性
        bool hasCritical = false;
        for (const auto& attr : attributes) {
            if (IsCriticalAttribute(attr.attributeId)) {
                hasCritical = true;
                break;
            }
        }

        if (hasCritical) {
            ExecutePersistence(task);
        } else {
            priorityQueues[priority].push(task);
        }

    // 立即保存所有属性
    void SaveAllAttributes(uint32_t playerId) {
        Player* player = PlayerManager::Instance()->GetPlayer(playerId);
        if (!player) return;

        auto allAttrs = player->GetAttributeContainer()->GetAllAttributes();

        PersistenceTask task;
        task.playerId = playerId;
        task.type = PERSIST_FULL_SAVE;
        task.timestamp = GetCurrentTime();
        task.priority = 9;  // 最高优先级

        for (const auto& pair : allAttrs) {
            auto def = AttributeDefManager::Instance()->GetDefinition(pair.first);
            if (def && def->isSaveable) {

                data.attributeId = pair.first;
                data.value = pair.second;
                task.attributes.push_back(data);

    bool LoadPlayerAttributes(uint32_t playerId,

            std::string sql = "SELECT attribute_id, value FROM player_attributes "
                             "WHERE player_id = ? AND deleted = 0";

private:
    PersistenceStats persistenceStats;

    // 持久化循环
    void PersistenceLoop() {
        while (running) {
            bool hasTask = false;

            // 按优先级处理任务
            for (int priority = 9; priority >= 0; --priority) {
                auto& queue = priorityQueues[priority];
                if (!queue.empty()) {
                    hasTask = true;
                    PersistenceTask task = queue.front();
                    queue.pop();
                    ExecutePersistence(task);
                }
            }

            // 定期刷新缓冲
            uint64_t now = GetCurrentTime();
            if (now - lastFlushTime > FLUSH_INTERVAL) {
                FlushAllBuffers();
                lastFlushTime = now;
            }

            if (!hasTask) {
                std::this_thread::sleep_for(std::chrono::milliseconds(10));
            }
        }

        // 退出时刷新所有数据
        FlushAllBuffers();
    }

    // 执行持久化
    void ExecutePersistence(const PersistenceTask& task) {
        try {
            switch (task.type) {
                case PERSIST_UPDATE:
                    ExecuteUpdate(task);
                    break;
                case PERSIST_INSERT:
                    ExecuteInsert(task);
                    break;
                case PERSIST_DELETE:
                    ExecuteDelete(task);
                    break;
                case PERSIST_BATCH:
                    ExecuteBatch(task);
                    break;
                case PERSIST_FULL_SAVE:
                    ExecuteFullSave(task);
                    break;
            }

            persistenceStats.totalSaves++;
            persistenceStats.totalAttributes += task.attributes.size();

        } catch (const std::exception& e) {
            LOG_ERROR("Persistence error for player %u: %s",
                      task.playerId, e.what());
            persistenceStats.failedSaves++;
        }
    }

    // 执行更新
    void ExecuteUpdate(const PersistenceTask& task) {
        for (const auto& attr : task.attributes) {
            std::string sql = "UPDATE player_attributes "
                             "SET value = ?, last_modified = ? "
                             "WHERE player_id = ? AND attribute_id = ?";

            auto stmt = DatabaseManager::Instance()->Prepare(sql);
            stmt->Bind(0, attr.value);
            stmt->Bind(1, task.timestamp);
            stmt->Bind(2, task.playerId);
            stmt->Bind(3, attr.attributeId);

            if (stmt->Execute() == 0) {
                // 没有更新行，可能是新属性，尝试插入
                ExecuteInsertAttribute(task.playerId, attr, task.timestamp);
            }
        }
    }

    // 执行批量更新
    void ExecuteBatch(const PersistenceTask& task) {

        auto transaction = DatabaseManager::Instance()->BeginTransaction();

        for (const auto& attr : task.attributes) {
            std::string sql = "INSERT INTO player_attributes "
                             "(player_id, attribute_id, value, last_modified) "
                             "VALUES (?, ?, ?, ?) "
                             "ON DUPLICATE KEY UPDATE "
                             "value = VALUES(value), "
                             "last_modified = VALUES(last_modified)";

            auto stmt = DatabaseManager::Instance()->Prepare(sql);
            stmt->Bind(0, task.playerId);
            stmt->Bind(1, attr.attributeId);
            stmt->Bind(2, attr.value);
            stmt->Bind(3, task.timestamp);

            stmt->Execute();
        }

        // 提交事务
        transaction->Commit();

        persistenceStats.batchSaves++;
    }

    // 执行完整保存
    void ExecuteFullSave(const PersistenceTask& task) {
        // 开始事务
        auto transaction = DatabaseManager::Instance()->BeginTransaction();

        // 先删除旧记录（标记为已删除）
        std::string deleteSql = "UPDATE player_attributes "
                                "SET deleted = 1, last_modified = ? "
                                "WHERE player_id = ?";
        auto deleteStmt = DatabaseManager::Instance()->Prepare(deleteSql);
        deleteStmt->Bind(0, task.timestamp);
        deleteStmt->Bind(1, task.playerId);
        deleteStmt->Execute();

        // 插入新记录
        std::string insertSql = "INSERT INTO player_attributes "
                                "(player_id, attribute_id, value, last_modified) "
                                "VALUES (?, ?, ?, ?)";
        auto insertStmt = DatabaseManager::Instance()->Prepare(insertSql);

        for (const auto& attr : task.attributes) {
            insertStmt->Bind(0, task.playerId);
            insertStmt->Bind(1, attr.attributeId);
            insertStmt->Bind(2, attr.value);
            insertStmt->Bind(3, task.timestamp);
            insertStmt->Execute();
        }

        // 提交事务
        transaction->Commit();

        persistenceStats.fullSaves++;
    }

    // 刷新玩家缓冲
    void FlushPlayerBuffer(uint32_t playerId) {
        std::lock_guard<std::mutex> lock(bufferMutex);

        auto it = updateBuffer.find(playerId);
        if (it != updateBuffer.end() && !it->second.empty()) {
            PersistenceTask task;
            task.playerId = playerId;
            task.type = PERSIST_BATCH;
            task.timestamp = GetCurrentTime();
            task.priority = 7;
            task.attributes = std::move(it->second);

            updateBuffer.erase(it);

            // 直接执行，不放入队列
            ExecutePersistence(task);
        }
    }

    // 刷新所有缓冲
    void FlushAllBuffers() {
        std::lock_guard<std::mutex> lock(bufferMutex);

        for (auto& pair : updateBuffer) {
            if (!pair.second.empty()) {
                PersistenceTask task;
                task.playerId = pair.first;
                task.type = PERSIST_BATCH;
                task.timestamp = GetCurrentTime();
                task.priority = 5;
                task.attributes = std::move(pair.second);

                ExecutePersistence(task);
            }
        }

        updateBuffer.clear();
    }
```

```cpp
class AttributeCache {
    // 多级缓存结构
    struct CacheEntry {
        int64_t value;
        uint64_t lastUpdate;
        uint64_t accessCount;
        uint64_t lastAccess;
        bool isDirty;
        uint32_t version;
    };

    // L1缓存：内存缓存（热点数据）
    LRUCache<uint32_t, CacheEntry> l1Cache{10000};

    // L2缓存：Redis缓存（跨服务器共享）
    RedisAttributeCache l2Cache;

    // 缓存统计
    std::atomic<uint64_t> l1Hits{0};
    std::atomic<uint64_t> l1Misses{0};
    std::atomic<uint64_t> l2Hits{0};
    std::atomic<uint64_t> l2Misses{0};

public:
    // 获取属性值（带缓存）
    int64_t GetAttribute(uint32_t playerId, uint32_t attributeId) {
        // L1缓存查找
        uint64_t cacheKey = BuildCacheKey(playerId, attributeId);
        auto entry = l1Cache.Get(cacheKey);

        if (entry) {
            l1Hits++;
            entry->accessCount++;
            entry->lastAccess = GetCurrentTime();
            return entry->value;
        }

        l1Misses++;

        // L2缓存查找
        int64_t value;
        if (l2Cache.Get(playerId, attributeId, value)) {
            l2Hits++;

            // 回填L1缓存
            CacheEntry newEntry;
            newEntry.value = value;
            newEntry.lastUpdate = GetCurrentTime();
            newEntry.accessCount = 1;
            newEntry.lastAccess = GetCurrentTime();
            newEntry.isDirty = false;
            newEntry.version = 1;

            l1Cache.Put(cacheKey, newEntry);
            return value;
        }

        l2Misses++;

        // 缓存未命中，从数据库加载
        return LoadFromDatabase(playerId, attributeId);
    }

    // 设置属性值（带缓存）
    void SetAttribute(uint32_t playerId, uint32_t attributeId, int64_t value) {
        uint64_t cacheKey = BuildCacheKey(playerId, attributeId);

        // 更新L1缓存
        CacheEntry entry;
        entry.value = value;
        entry.lastUpdate = GetCurrentTime();
        entry.accessCount = 1;
        entry.lastAccess = GetCurrentTime();
        entry.isDirty = true;
        entry.version = 1;

        l1Cache.Put(cacheKey, entry);

        // 异步更新L2缓存
        l2Cache.SetAsync(playerId, attributeId, value);
    }

    // 批量预加载
    void PreloadAttributes(uint32_t playerId, const std::vector<uint32_t>& attributeIds) {
        std::vector<std::future<void>> futures;

        for (uint32_t attrId : attributeIds) {
            futures.push_back(std::async(std::launch::async, [this, playerId, attrId]() {
                GetAttribute(playerId, attrId);  // 触发缓存加载
            }));
        }

        // 等待所有预加载完成
        for (auto& future : futures) {
            future.wait();
        }
    }

    void InvalidateAttribute(uint32_t playerId, uint32_t attributeId) {

        stats.l1Hits = l1Hits.load();

        // 从数据库加载属性值
        int64_t value = 0;

        try {
            std::string sql = "SELECT value FROM player_attributes "
                             "WHERE player_id = ? AND attribute_id = ?";

            stmt->Bind(0, playerId);
            stmt->Bind(1, attributeId);

            auto result = stmt->ExecuteQuery();
            if (result && result->Next()) {
                value = result->GetInt64(0);
            } else {
                // 使用默认值
                auto def = AttributeDefManager::Instance()->GetDefinition(attributeId);
                value = def ? def->defaultValue : 0;
            }

            // 更新缓存
            SetAttribute(playerId, attributeId, value);

        } catch (const std::exception& e) {
            LOG_ERROR("Failed to load attribute %u for player %u: %s",
                      attributeId, playerId, e.what());
        }

        return value;
    }
```

```cpp
// AttributeMemoryPool.h
template<typename T>
class AttributeMemoryPool {
    struct Block {
        T* data;
        size_t size;
        size_t used;
        std::vector<bool> allocated;
    };

    std::vector<std::unique_ptr<Block>> blocks;
    size_t blockSize;
    std::mutex poolMutex;

    // 空闲列表
    std::stack<T*> freeList;
    std::mutex freeListMutex;

    // 统计信息
    std::atomic<size_t> totalAllocated{0};
    std::atomic<size_t> totalFreed{0};
    std::atomic<size_t> currentUsed{0};

public:
    AttributeMemoryPool(size_t blockSize = 1024) : blockSize(blockSize) {}

    // 分配内存
    T* Allocate() {
        // 先尝试从空闲列表获取
        {
            std::lock_guard<std::mutex> lock(freeListMutex);
            if (!freeList.empty()) {
                T* ptr = freeList.top();
                freeList.pop();
                totalAllocated++;
                currentUsed++;
                return ptr;
            }
        }

        // 空闲列表为空，需要分配新块
        std::lock_guard<std::mutex> lock(poolMutex);

        // 查找有空闲空间的块
        for (auto& block : blocks) {
            if (block->used < block->size) {
                size_t index = FindFreeSlot(block);
                block->allocated[index] = true;
                block->used++;

                T* ptr = &block->data[index];
                new(ptr) T();  // 构造对象

                totalAllocated++;
                currentUsed++;
                return ptr;
            }
        }

        // 没有空闲块，创建新块
        auto newBlock = std::make_unique<Block>();
        newBlock->size = blockSize;
        newBlock->used = 1;
        newBlock->data = static_cast<T*>(::operator new[](blockSize * sizeof(T)));
        newBlock->allocated.resize(blockSize, false);
        newBlock->allocated[0] = true;

        T* ptr = &newBlock->data[0];
        new(ptr) T();

        blocks.push_back(std::move(newBlock));

        totalAllocated++;
        currentUsed++;
        return ptr;
    }

    // 释放内存
    void Deallocate(T* ptr) {
        if (!ptr) return;

        // 调用析构函数
        ptr->~T();

        // 加入空闲列表
        {
            std::lock_guard<std::mutex> lock(freeListMutex);
            freeList.push(ptr);
        }

        totalFreed++;
        currentUsed--;

        // 如果空闲列表过大，可以释放一些块
        if (freeList.size() > blockSize * 2) {
            ShrinkPool();
        }
    }

    // 获取统计信息
    MemoryPoolStats GetStats() const {
        MemoryPoolStats stats;
        stats.totalAllocated = totalAllocated.load();
        stats.totalFreed = totalFreed.load();
        stats.currentUsed = currentUsed.load();
        stats.blockCount = blocks.size();
        stats.blockSize = blockSize;
        stats.freeListSize = freeList.size();
        return stats;
    }

private:
    size_t FindFreeSlot(Block* block) {
        for (size_t i = 0; i < block->size; ++i) {
            if (!block->allocated[i]) {
                return i;
            }
        }
        return block->size;  // 没有空闲槽
    }
```
