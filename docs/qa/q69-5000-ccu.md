## 二、网络优化

### 2.1 C10K 解决方案

```cpp
// 使用 Epoll 解决 C10K

#include <sys/epoll.h>

class EpollServer {
public:
    bool start(int port) {
        // 创建 epoll
        epollFd_ = epoll_create1(0);
        if (epollFd_ < 0) return false;

        // 创建监听 socket
        listenFd_ = createListenSocket(port);
        if (listenFd_ < 0) return false;

        // 添加到 epoll
        struct epoll_event ev;
        ev.events = EPOLLIN | EPOLLET;  // 边缘触发
        ev.data.fd = listenFd_;
        epoll_ctl(epollFd_, EPOLL_CTL_ADD, listenFd_, &ev);

        // 设置为非阻塞
        setNonBlocking(listenFd_);

        return true;
    }

    void run() {
        struct epoll_event events[MAX_EVENTS];

        while (running_) {
            // 等待事件
            int nfds = epoll_wait(epollFd_, events, MAX_EVENTS, timeout_);

            for (int i = 0; i < nfds; ++i) {
                if (events[i].data.fd == listenFd_) {
                    acceptNewConnections();
                } else {
                    handleClientEvent(events[i]);
                }
            }
        }
    }

private:
    void acceptNewConnections() {
        while (true) {
            int clientFd = accept(listenFd_, nullptr, nullptr);
            if (clientFd < 0) {
                if (errno == EAGAIN || errno == EWOULDBLOCK) {
                    break;  // 没有更多连接
                }
                continue;
            }

            setNonBlocking(clientFd);

            // 添加到 epoll
            struct epoll_event ev;
            ev.events = EPOLLIN | EPOLLET | EPOLLRDHUP;
            ev.data.fd = clientFd;
            epoll_ctl(epollFd_, EPOLL_CTL_ADD, clientFd, &ev);

            connections_[clientFd] = std::make_unique<Connection>(clientFd);
        }
    }

    static constexpr int MAX_EVENTS = 1024;
    int epollFd_;
    int listenFd_;
    std::unordered_map<int, std::unique_ptr<Connection>> connections_;
};
```

### 2.2 广播优化

```cpp
// 优化 AOI 广播

class BroadcastOptimizer {
public:
    // 批量发送广播
    void broadcast(const Vector3& center, float radius, const Message& msg) {
        // 1. 获取范围内的玩家
        auto entities = aoi_.query(center, radius);

        if (entities.empty()) return;

        // 2. 序列化一次
        SerializedMessage serialized = serialize(msg);

        // 3. 按连接分组
        std::unordered_map<int, std::vector<int>> byServer;
        for (auto* entity : entities) {
            int serverId = getServerId(entity);
            byServer[serverId].push_back(entity->getPlayerId());
        }

        // 4. 批量发送
        for (const auto& [serverId, playerIds] : byServer) {
            sendBatch(serverId, playerIds, serialized);
        }
    }

private:
    void sendBatch(int serverId, const std::vector<int>& playerIds,
                   const SerializedMessage& msg) {
        // 批量发送到同一服务器
        BatchMessage batch;
        batch.playerIds = playerIds;
        batch.data = msg.data;
        batch.size = msg.size;

        sendToServer(serverId, batch);
    }

    AOIManager aoi_;
};
```

## 三、CPU 优化

### 3.1 多线程架构

```cpp
// 主线程 + 工作线程池

class ThreadedGameServer {
public:
    void start(int workerThreads = 4) {
        // 主线程处理网络 I/O
        mainThread_ = std::thread(&ThreadedGameServer::networkLoop, this);

        // 工作线程处理游戏逻辑
        for (int i = 0; i < workerThreads; ++i) {
            workers_.emplace_back(&ThreadedGameServer::workerLoop, this, i);
        }
    }

private:
    void networkLoop() {
        while (running_) {
            // 接收消息
            auto messages = receiveMessages();

            // 分发到工作线程
            for (auto& msg : messages) {
                int workerId = msg.playerId % workers_.size();
                workerQueues_[workerId].push(std::move(msg));
            }

            // 发送响应
            sendResponses();
        }
    }

    void workerLoop(int workerId) {
        auto& queue = workerQueues_[workerId];

        while (running_) {
            Message msg;
            if (queue.pop(msg)) {
                handleMessage(msg);
            } else {
                std::this_thread::sleep_for(std::chrono::microseconds(100));
            }
        }
    }

    std::thread mainThread_;
    std::vector<std::thread> workers_;
    std::vector<ThreadSafeQueue<Message>> workerQueues_;
};
```

### 3.2 无锁队列

```cpp
// 无锁队列减少锁竞争

template<typename T>
class LockFreeQueue {
public:
    bool push(T&& item) {
        Node* node = new Node{std::move(item), nullptr};

        while (true) {
            Node* last = tail_.load(std::memory_order_acquire);
            Node* next = last->next.load(std::memory_order_acquire);

            if (last == tail_.load(std::memory_order_acquire)) {
                if (next == nullptr) {
                    if (last->next.compare_exchange_weak(next, node,
                            std::memory_order_release,
                            std::memory_order_acquire)) {
                        tail_.compare_exchange_weak(last, node,
                            std::memory_order_release,
                            std::memory_order_acquire);
                        return true;
                    }
                } else {
                    tail_.compare_exchange_weak(last, next,
                        std::memory_order_release,
                        std::memory_order_acquire);
                }
            }
        }
    }

    bool pop(T& item) {
        while (true) {
            Node* first = head_.load(std::memory_order_acquire);
            Node* last = tail_.load(std::memory_order_acquire);
            Node* next = first->next.load(std::memory_order_acquire);

            if (first == head_.load(std::memory_order_acquire)) {
                if (first == last) {
                    if (next == nullptr) {
                        return false;  // 队列空
                    }
                    tail_.compare_exchange_weak(last, next,
                        std::memory_order_release,
                        std::memory_order_acquire);
                } else {
                    item = next->data;
                    if (head_.compare_exchange_weak(first, next,
                            std::memory_order_release,
                            std::memory_order_acquire)) {
                        delete first;
                        return true;
                    }
                }
            }
        }
    }

private:
    struct Node {
        T data;
        std::atomic<Node*> next;
    };

    std::atomic<Node*> head_;
    std::atomic<Node*> tail_;
};
```

## 四、内存优化

### 4.1 对象池

```cpp
// 玩家对象池

class PlayerPool {
public:
    PlayerPool(size_t capacity = 10000) {
        players_.reserve(capacity);
        freeIndices_.reserve(capacity);

        for (size_t i = 0; i < capacity; ++i) {
            players_.emplace_back(std::make_unique<Player>());
            freeIndices_.push_back(i);
        }
    }

    Player* acquire() {
        if (freeIndices_.empty()) {
            return nullptr;  // 池满了
        }

        size_t index = freeIndices_.back();
        freeIndices_.pop_back();

        Player* player = players_[index].get();
        player->reset();
        player->active_ = true;

        return player;
    }

    void release(Player* player) {
        if (!player || !player->active_) return;

        player->active_ = false;
        size_t index = player - players_[0].get();
        freeIndices_.push_back(index);
    }

private:
    std::vector<std::unique_ptr<Player>> players_;
    std::vector<size_t> freeIndices_;
};
```

```cpp
// 性能监控指标

struct PerformanceMetrics {
    // 连接指标
    std::atomic<int> currentConnections{0};
    std::atomic<int> peakConnections{0};

    // 消息指标
    std::atomic<uint64_t> messagesReceived{0};
    std::atomic<uint64_t> messagesSent{0};
    std::atomic<uint64_t> messageBytes{0};

    // CPU 指标
    std::atomic<double> cpuUsage{0};
    std::atomic<int> activeThreads{0};

    // 内存指标
    std::atomic<size_t> memoryUsed{0};
    std::atomic<size_t> memoryPeak{0};

    // 游戏指标
    std::atomic<int> entityCount{0};
    std::atomic<int> aoiUpdates{0};

    void printReport() {
        INFO("=== Performance Report ===");
        INFO("Connections: {}/{}", currentConnections.load(), peakConnections.load());
        INFO("Messages: {} sent, {} received",
             messagesSent.load(), messagesReceived.load());
        INFO("Memory: {} MB (peak: {} MB)",
             memoryUsed.load() / 1024 / 1024,
             memoryPeak.load() / 1024 / 1024);
        INFO("Entities: {}", entityCount.load());
    }
};
```
