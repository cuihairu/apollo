```cpp
// NetSelectReactor.h - IOCP反应器
class NetSelectReactor {
    HANDLE iocpHandle;
    std::vector<std::thread> workerThreads;

    // 启动IOCP线程池
    bool Start(int threadCount) {
        iocpHandle = CreateIoCompletionPort(INVALID_HANDLE_VALUE, NULL, 0, 0);

        for (int i = 0; i < threadCount; ++i) {
            workerThreads.emplace_back([this]() {
                DWORD bytesTransferred;
                ULONG_PTR completionKey;
                LPOVERLAPPED overlapped;

                while (GetQueuedCompletionStatus(iocpHandle,
                    &bytesTransferred, &completionKey, &overlapped, INFINITE)) {

                    // 处理网络事件
                    HandleNetworkEvent(completionKey, bytesTransferred, overlapped);
                }
            });
        }
    }
};
```

### 2.3 负载均衡策略

#### GameServer选择算法
```cpp
// GateServer/src/LoadBalancer.h
class LoadBalancer {
    struct ServerInfo {
        uint32_t serverId;
        int currentLoad;      // 当前负载
        int maxCapacity;      // 最大容量
        double weight;        // 权重
        bool isHealthy;       // 健康状态
    };

    std::map<uint32_t, ServerInfo> servers;

    // 选择负载最小的服务器
    uint32_t SelectBestServer() {
        uint32_t bestServer = 0;
        double minRatio = DBL_MAX;

        for (auto& pair : servers) {
            ServerInfo& info = pair.second;
            if (!info.isHealthy) continue;

            // 计算负载比例
            double ratio = (double)info.currentLoad / info.maxCapacity;

            // 考虑权重
            ratio /= info.weight;

            if (ratio < minRatio) {
                minRatio = ratio;
                bestServer = pair.first;
            }
        }

        return bestServer;
    }

    // 更新服务器负载
    void UpdateServerLoad(uint32_t serverId, int delta) {
        auto it = servers.find(serverId);
        if (it != servers.end()) {
            it->second.currentLoad += delta;

            // 负载过高时标记为不健康
            if (it->second.currentLoad > it->second.maxCapacity) {
                it->second.isHealthy = false;
            }
        }
    }
};
```

```cpp
// NetPacket.h - 数据包结构
#pragma pack(push, 1)
struct PacketHeader {
    uint16_t length;        // 数据包长度
    uint16_t message_id;    // 消息ID
    uint32_t sequence;      // 序列号
    uint32_t timestamp;     // 时间戳
    uint16_t checksum;      // 校验和
};

class NetPacket {
    PacketHeader header;
    char* data;
    int dataSize;

public:
    // 打包Protobuf消息
    template<typename T>
    bool PackMessage(uint16_t msgId, const T& message) {
        // 序列化消息
        std::string serialized;
        if (!ProtobufHelper<T>::Serialize(message, serialized)) {
            return false;
        }

        // 设置包头
        header.length = sizeof(PacketHeader) + serialized.size();
        header.message_id = msgId;
        header.sequence = GetNextSequence();
        header.timestamp = GetCurrentTimestamp();
        header.checksum = CalculateChecksum(serialized);

        // 复制数据
        dataSize = serialized.size();
        data = new char[dataSize];
        memcpy(data, serialized.c_str(), dataSize);

        return true;
    }

    // 解包到Protobuf消息
    template<typename T>
    bool UnpackMessage(T& message) const {
        if (!data || dataSize <= 0) {
            return false;
        }

        // 验证校验和
        if (!ValidateChecksum()) {
            return false;
        }

        // 反序列化
        return ProtobufHelper<T>::Deserialize(
            std::string(data, dataSize), message);
    }
};
#pragma pack(pop)
```

```cpp
// MessageDispatcher.h
class MessageDispatcher {
    using HandlerMap = std::unordered_map<uint16_t,
        std::shared_ptr<IMessageHandlerBase>>;

    HandlerMap handlers;

public:
    static MessageDispatcher* Instance() {
        static MessageDispatcher instance;
        return &instance;
    }

    // 注册处理器
    template<typename T>
    void RegisterHandler(uint16_t msgId, IMessageHandler<T>* handler) {
        // 包装处理器
        auto wrappedHandler = std::make_shared<WrappedHandler<T>>(handler);
        handlers[msgId] = wrappedHandler;

        LOG_INFO("Registered message handler: msgId=%d, type=%s",
                msgId, typeid(T).name());
    }

    // 分发消息
    bool DispatchMessage(uint32_t sessionId, NetPacket* packet) {
        auto it = handlers.find(packet->header.message_id);
        if (it == handlers.end()) {
            LOG_ERROR("No handler found for message: %d",
                     packet->header.message_id);
            return false;
        }

        // 根据消息类型分发
        return it->second->HandleMessage(sessionId, packet);
    }
};
```

## 5. 数据广播系统

### 5.1 广播系统架构

#### AOI（Area of Interest）管理
```cpp
// AOIManager.h
class AOIManager {
    struct AOIObject {
        uint32_t objectId;
        Vector3 position;
        float viewRadius;       // 视野半径
        std::set<uint32_t> watchers;  // 观察者列表
        std::set<uint32_t> watching;  // 正在观察的对象
    };

    // 使用空间索引提高性能
    std::unordered_map<uint32_t, AOIObject> objects;
    SpatialHashGrid spatialGrid;

public:
    // 更新对象位置
    void UpdatePosition(uint32_t objectId, const Vector3& newPos) {
        auto& obj = objects[objectId];
        Vector3 oldPos = obj.position;
        obj.position = newPos;

        // 获取旧的观察者
        auto oldWatchers = GetObjectsInRange(oldPos, obj.viewRadius);

        // 获取新的观察者
        auto newWatchers = GetObjectsInRange(newPos, obj.viewRadius);

        // 计算进入和离开视野的对象
        std::set<uint32_t> entered, left;

        // 进入视野的对象
        for (uint32_t watcherId : newWatchers) {
            if (oldWatchers.find(watcherId) == oldWatchers.end()) {
                entered.insert(watcherId);
            }
        }

        // 离开视野的对象
        for (uint32_t watcherId : oldWatchers) {
            if (newWatchers.find(watcherId) == newWatchers.end()) {
                left.insert(watcherId);
            }
        }

        // 发送进入视野通知
        if (!entered.empty()) {
            NotifyEnterView(objectId, entered);
        }

        // 发送离开视野通知
        if (!left.empty()) {
            NotifyLeaveView(objectId, left);
        }

        // 更新观察关系
        UpdateWatchingRelations(objectId, entered, left);
    }

    // 获取指定范围内的对象
    std::vector<uint32_t> GetObjectsInRange(const Vector3& pos, float radius) {
        return spatialGrid.Query(pos, radius);
    }

private:
    // 空间哈希网格实现
    class SpatialHashGrid {
        static const int GRID_SIZE = 100;  // 网格大小

        struct Cell {
            std::vector<uint32_t> objects;
        };

        std::unordered_map<uint64_t, Cell> grid;

    public:
        // 计算格子坐标
        uint64_t GetCellKey(const Vector3& pos) {
            int x = static_cast<int>(pos.x / GRID_SIZE);
            int y = static_cast<int>(pos.y / GRID_SIZE);
            return (static_cast<uint64_t>(x) << 32) | y;
        }

        // 查询范围内的对象
        std::vector<uint32_t> Query(const Vector3& center, float radius) {
            std::vector<uint32_t> result;

            int minCellX = static_cast<int>((center.x - radius) / GRID_SIZE);
            int maxCellX = static_cast<int>((center.x + radius) / GRID_SIZE);
            int minCellY = static_cast<int>((center.y - radius) / GRID_SIZE);
            int maxCellY = static_cast<int>((center.y + radius) / GRID_SIZE);

            for (int x = minCellX; x <= maxCellX; ++x) {
                for (int y = minCellY; y <= maxCellY; ++y) {
                    uint64_t key = (static_cast<uint64_t>(x) << 32) | y;
                    auto it = grid.find(key);
                    if (it != grid.end()) {
                        for (uint32_t objId : it->second.objects) {
                            // 精确距离计算
                            auto objPos = GetObjectPosition(objId);
                            if (Vector3::Distance(center, objPos) <= radius) {
                                result.push_back(objId);
                            }
                        }
                    }
                }
            }

            return result;
        }
    };
};
```

```cpp
// BroadcastMessage.h
enum BroadcastType {
    BROADCAST_AREA = 0,      // 区域广播（场景内）
    BROADCAST_AOI = 1,       // 视野广播（AOI内）
    BROADCAST_GUILD = 2,     // 公会广播
    BROADCAST_TEAM = 3,      // 队伍广播
    BROADCAST_WORLD = 4,     // 世界广播
    BROADCAST_SYSTEM = 5,    // 系统广播
    BROADCAST_CHANNEL = 6    // 频道广播
};

class BroadcastSystem {
public:
    // 区域广播
    template<typename T>
    void BroadcastToArea(uint32_t sceneId, const T& message,
                        uint32_t excludeId = 0) {
        Scene* scene = SceneManager::Instance()->GetScene(sceneId);
        if (!scene) return;

        for (Player* player : scene->GetPlayers()) {
            if (player->GetId() != excludeId) {
                SendToClient(player->GetSessionId(), message);
            }
        }
    }

    // AOI广播
    template<typename T>
    void BroadcastToAOI(Player* player, const T& message) {
        auto aoiManager = AOIManager::Instance();
        auto watchers = aoiManager->GetObjectsInRange(
            player->GetPosition(),
            player->GetViewRadius()
        );

        for (uint32_t watcherId : watchers) {
            Player* watcher = GetPlayer(watcherId);
            if (watcher) {
                SendToClient(watcher->GetSessionId(), message);
            }
        }
    }

    // 公会广播
    template<typename T>
    void BroadcastToGuild(uint32_t guildId, const T& message) {
        Guild* guild = GuildManager::Instance()->GetGuild(guildId);
        if (!guild) return;

        for (uint32_t memberId : guild->GetOnlineMembers()) {
            Player* member = GetPlayer(memberId);
            if (member) {
                SendToClient(member->GetSessionId(), message);
            }
        }
    }

    // 队伍广播
    template<typename T>
    void BroadcastToTeam(uint32_t teamId, const T& message) {
        Team* team = TeamManager::Instance()->GetTeam(teamId);
        if (!team) return;

        for (uint32_t memberId : team->GetMembers()) {
            Player* member = GetPlayer(memberId);
            if (member) {
                SendToClient(member->GetSessionId(), message);
            }
        }
    }

    // 延迟广播（批量处理）
    template<typename T>
    void BatchBroadcast(BroadcastType type, const T& message,
                       const std::vector<uint32_t>& targets) {
        // 添加到批量处理队列
        BatchBroadcastTask task;
        task.type = type;
        task.messageData = SerializeMessage(message);
        task.targets = targets;
        task.timestamp = GetCurrentTime();

        batchQueue.push(task);
    }

private:
    // 批量处理队列
    ThreadSafeQueue<BatchBroadcastTask> batchQueue;

    // 处理批量广播
    void ProcessBatchBroadcasts() {
        std::vector<BatchBroadcastTask> tasks;
        batchQueue.try_pop_bulk(tasks, 100);  // 每次处理100个

        // 按消息类型分组
        std::map<uint16_t, std::vector<BatchBroadcastTask>> groupedTasks;
        for (auto& task : tasks) {
            groupedTasks[task.messageId].push_back(task);
        }

        // 合并相同类型的消息
        for (auto& pair : groupedTasks) {
            auto& taskList = pair.second;
            if (taskList.size() > 1) {
                // 合并广播目标
                std::set<uint32_t> allTargets;
                for (auto& task : taskList) {
                    allTargets.insert(task.targets.begin(), task.targets.end());
                }

                // 发送合并后的广播
                SendMergedBroadcast(taskList[0], allTargets);
            } else {
                // 单独发送
                SendSingleBroadcast(taskList[0]);
            }
        }
    }
};
```

### 5.3 广播优化策略

#### 批量广播机制
```cpp
// BatchBroadcast.h
class BatchBroadcastManager {
    struct BroadcastBatch {
        uint16_t messageId;
        std::string messageData;
        std::vector<uint32_t> targets;
        uint64_t timestamp;
        int priority;  // 优先级
    };

    // 多级优先级队列
    std::array<ThreadSafeQueue<BroadcastBatch>, 4> priorityQueues;
    std::thread workerThread;
    std::atomic<bool> running{true};

public:
    BatchBroadcastManager() {
        workerThread = std::thread(&BatchBroadcastManager::WorkerLoop, this);
    }

    ~BatchBroadcastManager() {
        running = false;
        if (workerThread.joinable()) {
            workerThread.join();
        }
    }

    // 添加广播任务
    template<typename T>
    void AddBroadcastTask(const T& message,
                         const std::vector<uint32_t>& targets,
                         int priority = 0) {
        BroadcastBatch batch;
        batch.messageId = T::MSG_ID;
        batch.messageData = SerializeMessage(message);
        batch.targets = targets;
        batch.timestamp = GetCurrentTime();
        batch.priority = priority;

        // 根据优先级加入队列
        priorityQueues[priority].push(batch);
    }

private:
    void WorkerLoop() {
        while (running) {
            // 按优先级处理队列
            for (int i = 0; i < 4; ++i) {
                std::vector<BroadcastBatch> batches;
                priorityQueues[i].try_pop_bulk(batches, 50);

                if (!batches.empty()) {
                    ProcessBatches(batches);
                }
            }

            // 短暂休眠
            std::this_thread::sleep_for(std::chrono::milliseconds(10));
        }
    }

    void ProcessBatches(const std::vector<BroadcastBatch>& batches) {
        // 按消息ID分组
        std::map<uint16_t, std::vector<BroadcastBatch>> grouped;
        for (const auto& batch : batches) {
            grouped[batch.messageId].push_back(batch);
        }

        // 处理每种消息类型
        for (const auto& pair : grouped) {
            const auto& batchList = pair.second;

            // 合并目标
            std::set<uint32_t> allTargets;
            for (const auto& batch : batchList) {
```

### 6.1 序列号去重

#### SequenceWindow实现

```cpp
// SequenceWindow.h
class SequenceWindow {
    static const uint32_t WINDOW_SIZE = 1024;  // 滑动窗口大小

    struct SequenceEntry {
        bool received;
        uint64_t timestamp;
    };

    uint32_t baseSequence;           // 基础序列号
    std::vector<SequenceEntry> window;
    std::mutex mutex;

public:
    SequenceWindow() : baseSequence(0), window(WINDOW_SIZE) {
        for (auto& entry : window) {
            entry.received = false;
            entry.timestamp = 0;
        }
    }

    // 检查并标记序列号
    bool CheckAndMark(uint32_t sequence) {
        std::lock_guard<std::mutex> lock(mutex);

        // 如果是旧序列号，拒绝
        if (sequence < baseSequence) {
            return false;
        }

        // 如果超出窗口范围，滑动窗口
        if (sequence >= baseSequence + WINDOW_SIZE) {
            uint32_t shift = sequence - (baseSequence + WINDOW_SIZE) + 1;
            SlideWindow(shift);
        }

        // 检查是否已接收
        uint32_t index = sequence % WINDOW_SIZE;
        if (window[index].received) {
            return false;  // 重复包
        }

        // 标记为已接收
        window[index].received = true;
        window[index].timestamp = GetCurrentTime();

        return true;
    }

    // 清理过期的序列号
    void Cleanup() {
        std::lock_guard<std::mutex> lock(mutex);

        uint64_t now = GetCurrentTime();
        uint64_t expireTime = 5 * 60 * 1000;  // 5分钟过期

        // 找到最早未过期的序列号
        uint32_t newBase = baseSequence;
        for (uint32_t i = 0; i < WINDOW_SIZE; ++i) {
            uint32_t seq = baseSequence + i;
            uint32_t index = seq % WINDOW_SIZE;

            if (window[index].received &&
                (now - window[index].timestamp) > expireTime) {
                window[index].received = false;
                window[index].timestamp = 0;
                newBase = seq + 1;
            } else {
                break;
            }
        }

        if (newBase > baseSequence) {
            SlideWindow(newBase - baseSequence);
        }
    }

private:
    void SlideWindow(uint32_t shift) {
        for (uint32_t i = 0; i < shift; ++i) {
            uint32_t index = (baseSequence + i) % WINDOW_SIZE;
            window[index].received = false;
            window[index].timestamp = 0;
        }
        baseSequence += shift;
    }
};
```

```cpp
// MessageFingerprint.h
class MessageFingerprint {
    // CRC32表
    static uint32_t crc32Table[256];
    static bool tableInitialized;

    static void InitCRC32Table() {
        if (tableInitialized) return;

        uint32_t polynomial = 0xEDB88320;
        for (uint32_t i = 0; i < 256; ++i) {
            uint32_t crc = i;
            for (int j = 8; j > 0; --j) {
                crc = crc & 1 ? (crc >> 1) ^ polynomial : crc >> 1;
            }
            crc32Table[i] = crc;
        }
        tableInitialized = true;
    }

public:
    // 计算消息指纹
    static uint32_t Calculate(const char* data, size_t length) {
        InitCRC32Table();

        uint32_t crc = 0xFFFFFFFF;
        for (size_t i = 0; i < length; ++i) {
            crc = (crc >> 8) ^ crc32Table[(crc ^ data[i]) & 0xFF];
        }

        return crc ^ 0xFFFFFFFF;
    }

    // 计算Protobuf消息指纹
    template<typename T>
    static uint32_t Calculate(const T& message) {
        std::string serialized;
        if (!message.SerializeToString(&serialized)) {
            return 0;
        }

        // 只计算关键字段的指纹，排除时间戳等
        return CalculateWithExclusion(serialized);
    }

private:
    // 排除某些字段的指纹计算
    static uint32_t CalculateWithExclusion(const std::string& data) {
        // 解析Protobuf并排除时间戳字段
        // 这里简化实现，实际需要根据消息结构调整
        return Calculate(data.c_str(), data.length());
    }
};

// 去重缓存
class DuplicateCache {
    struct CacheEntry {
        uint32_t fingerprint;
        uint64_t timestamp;
        uint32_t count;  // 重复次数
    };

    static const uint32_t CACHE_SIZE = 10000;
    std::vector<CacheEntry> cache;
    std::mutex mutex;

public:
    DuplicateCache() : cache(CACHE_SIZE) {}

    // 检查是否重复
    bool IsDuplicate(uint32_t messageId, uint32_t fingerprint) {
        std::lock_guard<std::mutex> lock(mutex);

        uint32_t index = (messageId + fingerprint) % CACHE_SIZE;
        CacheEntry& entry = cache[index];

        uint64_t now = GetCurrentTime();
        uint64_t expireTime = 60 * 1000;  // 1分钟过期

        if (entry.fingerprint == fingerprint &&
            (now - entry.timestamp) < expireTime) {
            entry.count++;
            return true;
        }

        entry.fingerprint = fingerprint;
        entry.timestamp = now;
        entry.count = 1;

        return false;
    }

    // 清理过期缓存
    void Cleanup() {
        std::lock_guard<std::mutex> lock(mutex);

        uint64_t now = GetCurrentTime();
        uint64_t expireTime = 5 * 60 * 1000;  // 5分钟过期

        for (auto& entry : cache) {
            if ((now - entry.timestamp) > expireTime) {
                entry.fingerprint = 0;
                entry.timestamp = 0;
                entry.count = 0;
            }
        }
    }
};
```

```cpp
// PacketDeduplicator.h
class PacketDeduplicator {
    SequenceWindow seqWindow;
    DuplicateCache duplicateCache;
    std::thread cleanupThread;
    std::atomic<bool> running{true};

    // 去重统计
    std::atomic<uint64_t> totalPackets{0};
    std::atomic<uint64_t> duplicatePackets{0};
    std::atomic<uint64_t> sequenceDuplicates{0};
    std::atomic<uint64_t> fingerprintDuplicates{0};

public:
    PacketDeduplicator() {
        cleanupThread = std::thread(&PacketDeduplicator::CleanupLoop, this);
    }

    ~PacketDeduplicator() {
        running = false;
        if (cleanupThread.joinable()) {
            cleanupThread.join();
        }
    }

    // 检查数据包是否重复
    bool IsDuplicate(NetPacket* packet) {
        totalPackets++;

        // 1. 序列号去重
        if (!seqWindow.CheckAndMark(packet->header.sequence)) {
            sequenceDuplicates++;
            LOG_DEBUG("Duplicate packet by sequence: %u",
                     packet->header.sequence);
            return true;
        }

        // 2. 消息指纹去重（仅对关键消息）
        if (ShouldCheckFingerprint(packet->header.message_id)) {
            uint32_t fingerprint = MessageFingerprint::Calculate(
                packet->data, packet->dataSize);

            if (duplicateCache.IsDuplicate(packet->header.message_id,
                                          fingerprint)) {
                fingerprintDuplicates++;
                LOG_DEBUG("Duplicate packet by fingerprint: msgId=%u, fp=%u",
                         packet->header.message_id, fingerprint);
                return true;
            }
        }

        return false;
    }

    // 获取去重统计
    DeduplicationStats GetStats() const {
        DeduplicationStats stats;
        stats.totalPackets = totalPackets.load();
        stats.duplicatePackets = duplicatePackets.load();
        stats.sequenceDuplicates = sequenceDuplicates.load();
        stats.fingerprintDuplicates = fingerprintDuplicates.load();
        stats.duplicateRate = (double)duplicatePackets / totalPackets;

        return stats;
    }

private:
    // 判断是否需要指纹检查
    bool ShouldCheckFingerprint(uint16_t messageId) {
        // 以下类型消息需要指纹检查
        return messageId == MSG_ATTRIBUTE_UPDATE ||
               messageId == MSG_PLAYER_MOVE ||
               messageId == MSG_SKILL_CAST ||
               messageId == MSG_ITEM_USE;
    }

    // 清理循环
    void CleanupLoop() {
        while (running) {
            // 清理过期数据
            seqWindow.Cleanup();
            duplicateCache.Cleanup();

            // 输出统计信息
            OutputStats();

            // 每分钟清理一次
            std::this_thread::sleep_for(std::chrono::minutes(1));
        }
    }

    // 输出统计信息
    void OutputStats() {
        auto stats = GetStats();
        LOG_INFO("Deduplication Stats - Total: %llu, Duplicate: %llu, Rate: %.2f%%",
                 stats.totalPackets, stats.duplicatePackets,
                 stats.duplicateRate * 100);
    }
};
```

```cpp
// LockFreeDeduplicator.h
class LockFreeDeduplicator {
    // 使用原子操作的位图
    std::atomic<uint64_t> bitmap[64];  // 支持4096个序列号

    // 原子操作的序列号更新
    std::atomic<uint32_t> lastSequence{0};

public:
    bool CheckAndMarkAtomic(uint32_t sequence) {
        uint32_t diff = sequence - lastSequence.load();

        // 如果序列号太旧，直接拒绝
        if (diff > 4096) {
            return false;
        }

        // 计算位图位置
        uint32_t index = sequence % 4096;
        uint32_t wordIndex = index / 64;
        uint32_t bitIndex = index % 64;
        uint64_t mask = 1ULL << bitIndex;

        // 原子操作检查并设置位
        uint64_t old = bitmap[wordIndex].fetch_or(mask);

        // 如果位已经设置，说明是重复
        return (old & mask) != 0;
    }

    // 定期清理位图
    void CleanupBitmap() {
        uint32_t current = lastSequence.load();
        uint32_t cleanupThreshold = current - 2048;  // 保留最近2048个

        for (uint32_t i = 0; i < 64; ++i) {
            uint64_t mask = 0;
            for (uint32_t j = 0; j < 64; ++j) {
                uint32_t seq = i * 64 + j;
                if (seq < cleanupThreshold) {
                    mask |= (1ULL << j);
                }
            }

            // 清理旧位
            if (mask != 0) {
                bitmap[i].fetch_and(~mask);
            }
        }
    }
};
```

```cpp
class PipeSession {
    HANDLE pipeHandle;
    std::string pipeName;
    std::thread receiveThread;
    std::queue<PipeMessage> sendQueue;
    std::mutex sendMutex;

public:
    // 连接到管道
    bool Connect(const std::string& name) {
        pipeName = "\\\\.\\pipe\\" + name;
        pipeHandle = CreateFileA(pipeName.c_str(),
                                GENERIC_READ | GENERIC_WRITE,
                                0, NULL, OPEN_EXISTING,
                                FILE_FLAG_OVERLAPPED, NULL);

        if (pipeHandle == INVALID_HANDLE_VALUE) {
            return false;
        }

        receiveThread = std::thread(&PipeSession::ReceiveLoop, this);
        return true;
    }

    // 发送消息
    void SendMessage(const PipeMessage& msg) {
        std::lock_guard<std::mutex> lock(sendMutex);
        sendQueue.push(msg);
    }

private:
    void ReceiveLoop() {
        while (true) {
            PipeMessage msg;
            DWORD bytesRead;

            if (ReadFile(pipeHandle, &msg, sizeof(msg),
                        &bytesRead, NULL) && bytesRead == sizeof(msg)) {
                HandleMessage(msg);
            }
        }
    }

    void HandleMessage(const PipeMessage& msg) {
        // 根据消息类型处理
        switch (msg.type) {
            case MSG_TYPE_DB_REQUEST:
                HandleDBRequest(msg);
                break;
            case MSG_TYPE_WORLD_BROADCAST:
                HandleWorldBroadcast(msg);
                break;
            case MSG_TYPE_GUILD_UPDATE:
                HandleGuildUpdate(msg);
                break;
        }
    }
};
```

```cpp
// ServiceRegistry.h
class ServiceRegistry {
    struct ServiceInfo {
        std::string name;
        std::string host;
        int port;
        ServiceStatus status;
        uint64_t lastHeartbeat;
        int weight;
    };

    std::map<std::string, std::vector<ServiceInfo>> services;
    std::mutex registryMutex;
    std::thread heartbeatThread;

public:
    // 注册服务
    void RegisterService(const ServiceInfo& info) {
        std::lock_guard<std::mutex> lock(registryMutex);
        services[info.name].push_back(info);
        LOG_INFO("Service registered: %s:%d", info.name.c_str(), info.port);
    }

    // 发现服务
    std::vector<ServiceInfo> DiscoverServices(const std::string& name) {
        std::lock_guard<std::mutex> lock(registryMutex);
        auto it = services.find(name);
        if (it != services.end()) {
            return it->second;
        }
        return {};
    }

    // 获取健康的服务
    ServiceInfo GetHealthyService(const std::string& name) {
        auto svcs = DiscoverServices(name);

        // 筛选健康的服务
        std::vector<ServiceInfo> healthyServices;
        for (const auto& svc : svcs) {
            if (svc.status == SERVICE_HEALTHY &&
                IsServiceAlive(svc)) {
                healthyServices.push_back(svc);
            }
        }

        // 负载均衡选择
        return LoadBalanceSelect(healthyServices);
    }

private:
    // 检查服务是否存活
    bool IsServiceAlive(const ServiceInfo& info) {
        uint64_t now = GetCurrentTime();
        return (now - info.lastHeartbeat) < 30000;  // 30秒超时
    }

    // 负载均衡选择
    ServiceInfo LoadBalanceSelect(const std::vector<ServiceInfo>& services) {
        // 加权随机选择
        int totalWeight = 0;
        for (const auto& svc : services) {
            totalWeight += svc.weight;
        }

        int random = GetRandom(0, totalWeight - 1);
        int currentWeight = 0;

        for (const auto& svc : services) {
            currentWeight += svc.weight;
            if (random < currentWeight) {
                return svc;
            }
        }

        return services[0];  // 默认返回第一个
    }
};
```
